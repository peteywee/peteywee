import AdmZip from "adm-zip";
import pdfParse    from "pdf-parse";
import yaml        from "js-yaml";
import { JSDOM }   from "jsdom";

async function extractText(file) {
  const buf = Buffer.from(file.content_base64, "base64");
  switch (file.filetype) {
    case "zip":
      const zip = new AdmZip(buf);
      return zip.getEntries()
        .map(e => e.getData().toString("utf8"))
        .join("\n\n–––\n\n");
    case "pdf":
      const data = await pdfParse(buf);
      return data.text;
    case "html":
      const dom = new JSDOM(buf.toString("utf8"));
      return dom.window.document.body.textContent;
    case "yml":
      return yaml.load(buf.toString("utf8"));
    default:  // json, text, makefile, dockerfile, terraform
      return buf.toString("utf8");
  }
}

3. Backend handler: decode, extract, chunk, summarize
Implement your /summarize_documents endpoint (Node.js example):

import AdmZip    from "adm-zip";
import pdfParse  from "pdf-parse";
import { JSDOM } from "jsdom";
import yaml      from "js-yaml";
import OpenAI    from "openai";
const openai = new OpenAI();

async function extractText(file) {
  const buf = Buffer.from(file.content_base64, "base64");
  switch (file.filetype) {
    case "zip":
      return new AdmZip(buf).getEntries()
        .map(e => e.getData().toString("utf8")).join("\n\n–––\n\n");
    case "pdf":
      return (await pdfParse(buf)).text;
    case "html":
      return new JSDOM(buf.toString("utf8")).window.document.body.textContent;
    case "yml":
      return yaml.dump(yaml.load(buf.toString("utf8")));
    default:  // json, text, makefile, dockerfile, terraform
      return buf.toString("utf8");
  }
}

function chunkText(text, maxChars = 3000) {
  const chunks = [];
  for (let i = 0; i < text.length; i += maxChars) {
    chunks.push(text.slice(i, i + maxChars));
  }
  return chunks;
}

async function summarizeText(text) {
  // A single-chunk summary call
  const res = await openai.chat.completions.create({
    model: "gpt-4o-mini",
    messages: [
      { role: "system",  content: "You are a helpful summarizer." },
      { role: "user",    content: `Please provide a concise summary of the following text:\n\n${text}` }
    ]
  });
  return res.choices[0].message.content.trim();
}

export async function handleSummarizeDocuments(req, res) {
  const { files } = req.body;
  const individual = [];

  for (const f of files) {
    const fullText = await extractText(f);
    const chunks   = chunkText(fullText);
    const summaries = [];
    for (const c of chunks) {
      summaries.push(await summarizeText(c));
    }
    // Merge chunk summaries into one
    const fileSummary = summaries.length > 1
      ? await summarizeText(summaries.join("\n\n"))
      : summaries[0];
    individual.push({ filename: f.filename, summary: fileSummary });
  }

  // Overall summary across files
  const allSummaries = individual.map(i => `– ${i.filename}:\n${i.summary}`).join("\n\n");
  const overall = await summarizeText(`Here are individual file summaries:\n\n${allSummaries}\n\nPlease provide an executive summary of all documents.`);

  return res.json({ individual_summaries: individual, overall_summary: overall });
}

3. Defining Endpoints & URLs in Cloudflare

DNS records pointing your hostname to an IP or CNAME.
