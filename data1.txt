from pathlib import Path
import os

# Security Enforcement Agent (SEA) script
sea_agent_script = """#!/usr/bin/env bash
set -euo pipefail

LOG_DIR="$HOME/.assistant/security"
mkdir -p "$LOG_DIR"
LOG_FILE="$LOG_DIR/sea_$(date +%Y%m%d).log"

echo "[SEA] Security Enforcement Agent started." | tee -a "$LOG_FILE"

# Watch for common security violations
check_permissions() {
  for file in "$HOME/.env" "$HOME/.ssh/id_rsa" "$HOME/.netrc"; do
    [ -f "$file" ] || continue
    perms=$(stat -c %a "$file")
    if [[ "$perms" != "600" && "$perms" != "644" ]]; then
      echo "[SEA] [ALERT] Insecure permissions on $file: $perms" | tee -a "$LOG_FILE"
      chmod 600 "$file"
      echo "[SEA] [ACTION] Permissions on $file fixed to 600" | tee -a "$LOG_FILE"
    fi
  done
}

check_cron_backdoors() {
  crontabs=$(find /var/spool/cron/crontabs -type f 2>/dev/null || true)
  for cron in $crontabs; do
    grep -q "curl" "$cron" && echo "[SEA] [ALERT] Suspicious curl command in $cron" | tee -a "$LOG_FILE"
  done
}

check_sudoers() {
  grep -q "ALL=(ALL) NOPASSWD:ALL" /etc/sudoers && echo "[SEA] [ALERT] Unsafe sudoers config!" | tee -a "$LOG_FILE"
}

monitor_file_changes() {
  echo "[SEA] Watching sensitive paths..." | tee -a "$LOG_FILE"
  inotifywait -m -e modify,create,delete $HOME/.ssh $HOME/.env /etc/sudoers /etc/passwd >> "$LOG_FILE" 2>&1 &
}

# Execute checks
check_permissions
check_cron_backdoors
check_sudoers
monitor_file_changes
"""
