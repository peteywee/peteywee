#!/usr/bin/env bash
set -e

owner="$TF_VAR_github_owner"
repos=("${REPOSITORIES[@]}")
root="$HOME/git-root"
skeleton="$root/tools/skeleton"

echo "[+] Bootstrapping repositories for owner: $owner"

for repo in "${repos[@]}"; do
  echo "[+] Processing $repo"
  tmp=$(mktemp -d)
  cp -R "$skeleton/" "$tmp/$repo"
  cd "$tmp/$repo"
  git init
  gh repo create "$owner/$repo" --private --confirm
  git remote add origin git@github.com:$owner/$repo.git
  git add .
  git commit -m "chore: initial skeleton"
  git push -u origin main
  cd - >/dev/null
  rm -rf "$tmp"
done

echo "[+] Bootstrap complete."

set -e

              "docker-vscode-setup" "ssh-setup" "firebase-spa" "conmon-build" \
              "ci-checks" "workspace-structure")

ROOT="$HOME/git-root"
ARCH_DIR="$(cd "$(dirname "$0")" && pwd)"

log()  { echo "[+] $1"; }
error(){ echo "[!] $1" >&2; exit 1; }

log "Checking dependencies..."
for cmd in git curl gpg terraform gh; do
  if ! command -v "$cmd" &>/dev/null; then
    log "Installing $cmd..."
    if [ "$cmd" == "terraform" ]; then
      curl -fsSL https://apt.releases.hashicorp.com/gpg \
        | sudo gpg --dearmor -o /usr/share/keyrings/hashicorp-archive-keyring.gpg
      echo "deb [signed-by=/usr/share/keyrings/hashicorp-archive-keyring.gpg] \
        https://apt.releases.hashicorp.com $(lsb_release -cs) main" \
        | sudo tee /etc/apt/sources.list.d/hashicorp.list
      sudo apt-get update -y
      sudo apt-get install -y terraform
    else
      sudo apt-get update -y
      sudo apt-get install -y "$cmd"
    fi
  fi
done

log "Authenticating GitHub CLI..."
echo "$GITHUB_TOKEN" | gh auth login --with-token || error "GitHub auth failed"

log "Scaffolding directories..."
for dir in \
  "$ROOT/modules/color_system" \
  "$ROOT/modules/vpc" \
  "$ROOT/modules/db" \
  "$ROOT/projects/billing/dev" \
  "$ROOT/projects/billing/prod" \
  "$ROOT/projects/analytics/dev" \
  "$ROOT/projects/color-system-bootstrap" \
  "$ROOT/tools/skeleton" \
  "$ROOT/tools/manifests"; do
  if [ ! -d "$dir" ]; then
    mkdir -p "$dir"
    log "Created $dir"
  else
    log "Exists  $dir"
  fi
done

log "Copying module templates..."
if [ ! -f "$ROOT/modules/color_system/main.tf" ]; then
  cp "$ARCH_DIR/templates/color_system/main.tf" \
    "$ROOT/modules/color_system/main.tf"
  log "Copied color_system/main.tf"
fi

log "Installing OS packages & runtimes..."
sudo apt-get update -y
xargs -a "$ARCH_DIR/manifests/packages.txt" sudo apt-get install -y
sudo apt-get install -y docker.io nodejs python3-pip

log "Bootstrapping GitHub repositories..."
for repo in "${REPOS[@]}"; do
  log "→ $repo"
  tmp=$(mktemp -d)
  cp -R "$ROOT/tools/skeleton" "$tmp/$repo"
  cd "$tmp/$repo"
  git init
  gh repo create "$TF_VAR_github_owner/$repo" --private --confirm || error "$repo creation failed"
  git remote add origin git@github.com:$TF_VAR_github_owner/$repo.git
  git add .
  git commit -m "chore: initial skeleton"
  git push -u origin main || error "$repo push failed"
  cd - >/dev/null
  rm -rf "$tmp"
done

log "Applying Terraform (repos & modules..."
cd "$ROOT"
terraform init -input=false
terraform apply -auto-approve || error "Terraform apply failed"

echo_green() { echo -e "\033[32m$1\033[0m"; }

# Prompt for a secret (hidden input)
prompt_secret() {
  local var_name=$1
  local prompt_msg=$2
  # If already stored, allow skipping
  local existing
  existing=$(grep "^${var_name}=" "$SECRETS_FILE" 2>/dev/null || true)
  if [[ -n "$existing" ]]; then
    read -p "$prompt_msg [hidden] (Enter=keep): " -s input
    echo
    [[ -z "$input" ]] && return
  else
    read -p "$prompt_msg [hidden]: " -s input
    echo

  chmod 600 "$SECRETS_FILE"
}

prompt_value() {
  local var_name=$1
  local prompt_msg=$2
  local default_val=$3

    input=${input:-$existing}

    read -p "$prompt_msg${default_val:+ [$default_val]}: " input
    input=${input:-$default_val}

  sed -i.bak "/^${var_name}=/d" "$SECRETS_FILE" 2>/dev/null || true
  echo "${var_name}=${input}" >> "$SECRETS_FILE"
  chmod 600 "$SECRETS_FILE"
}
