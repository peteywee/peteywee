#!/bin/bash
# knowledge_base_monitor.sh
#
# This script runs as a cron job to monitor the knowledge base,
# perform regular backups, security audits, and maintenance tasks.

set -e  # Exit immediately if a command exits with a non-zero status

# Configuration
KB_DIR="${KB_DIR:-$HOME/knowledge_base}"
KB_DB="${KB_DIR}/data/knowledge_base.db"
KB_BACKUP_DIR="${KB_DIR}/backups"
KB_LOG_DIR="${KB_DIR}/logs"
KB_CONFIG="${KB_DIR}/config/kb_config.json"
LOG_FILE="${KB_LOG_DIR}/monitor_$(date +%Y%m%d).log"

# Maximum number of backups to keep
MAX_BACKUPS=30

# Maximum age of logs in days
MAX_LOG_AGE=90

# ANSI color codes
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[0;33m'
BLUE='\033[0;34m'
NC='\033[0m' # No Color

# Logging functions
log() {
    echo -e "${BLUE}[$(date '+%Y-%m-%d %H:%M:%S')] ${GREEN}$1${NC}" | tee -a "${LOG_FILE}"
}

error() {
    echo -e "${RED}[ERROR] [$(date '+%Y-%m-%d %H:%M:%S')] $1${NC}" | tee -a "${LOG_FILE}" >&2
}

warn() {
    echo -e "${YELLOW}[WARNING] [$(date '+%Y-%m-%d %H:%M:%S')] $1${NC}" | tee -a "${LOG_FILE}"
}

# Ensure log directory exists
mkdir -p "${KB_LOG_DIR}"

# Create the log file if it doesn't exist
touch "${LOG_FILE}"

# Check if knowledge base exists
check_kb_exists() {
    if [ ! -d "${KB_DIR}" ]; then
        error "Knowledge base directory ${KB_DIR} not found"
        exit 1
    fi
    
    if [ ! -f "${KB_DB}" ]; then
        error "Knowledge base database ${KB_DB} not found"
        exit 1
    }
    
    log "Knowledge base exists at ${KB_DIR}"
}

# Activate virtual environment
activate_venv() {
    if [ -f "${KB_DIR}/venv/bin/activate" ]; then
        source "${KB_DIR}/venv/bin/activate"
        log "Virtual environment activated"
    else
        error "Virtual environment not found at ${KB_DIR}/venv"
        exit 1
    fi
}

# Create a backup of the knowledge base
create_backup() {
    log "Creating backup of knowledge base..."
    
    # Ensure backup directory exists
    mkdir -p "${KB_BACKUP_DIR}"
    
    # Create backup with timestamp
    backup_file="${KB_BACKUP_DIR}/knowledge_base_$(date +%Y%m%d_%H%M%S).db"
    sqlite3 "${KB_DB}" ".backup '${backup_file}'"
    
    # Check if backup was successful
    if [ $? -eq 0 ]; then
        log "Backup created successfully: ${backup_file}"
    else
        error "Backup failed"
        return 1
    fi
    
    # Compress backup to save space
    gzip "${backup_file}"
    log "Backup compressed: ${backup_file}.gz"
    
    return 0
}

# Clean up old backups
cleanup_old_backups() {
    log "Cleaning up old backups..."
    
    # Count the number of backups
    backup_count=$(ls -1 "${KB_BACKUP_DIR}"/knowledge_base_*.db.gz 2>/dev/null | wc -l)
    
    # If there are more backups than the maximum, delete the oldest ones
    if [ "${backup_count}" -gt "${MAX_BACKUPS}" ]; then
        # Calculate how many backups to delete
        delete_count=$((backup_count - MAX_BACKUPS))
        
        log "Found ${backup_count} backups, keeping ${MAX_BACKUPS}, deleting ${delete_count}"
        
        # Get the oldest backups
        oldest_backups=$(ls -1t "${KB_BACKUP_DIR}"/knowledge_base_*.db.gz | tail -n "${delete_count}")
        
        # Delete the oldest backups
        for backup in ${oldest_backups}; do
            rm "${backup}"
            log "Deleted old backup: ${backup}"
        done
    else
        log "Found ${backup_count} backups, no cleanup needed"
    fi
}

# Clean up old logs
cleanup_old_logs() {
    log "Cleaning up old logs..."
    
    # Find and delete logs older than MAX_LOG_AGE days
    find "${KB_LOG_DIR}" -name "*.log" -type f -mtime +${MAX_LOG_AGE} -delete
    
    log "Old logs cleaned up"
}

# Perform database optimization
optimize_database() {
    log "Optimizing database..."
    
    # Run VACUUM to rebuild the database file
    sqlite3 "${KB_DB}" "VACUUM;"
    
    # Run ANALYZE to update statistics
    sqlite3 "${KB_DB}" "ANALYZE;"
    
    log "Database optimization complete"
}

# Check database integrity
check_integrity() {
    log "Checking database integrity..."
    
    # Run integrity check
    integrity=$(sqlite3 "${KB_DB}" "PRAGMA integrity_check;")
    
    if [ "${integrity}" == "ok" ]; then
        log "Database integrity check passed"
    else
        error "Database integrity check failed: ${integrity}"
        return 1
    fi
    
    return 0
}

# Run security audit
run_security_audit() {
    log "Running security audit..."
    
    # Run security audit script
    python "${KB_DIR}/scripts/kb_security.py" --db "${KB_DB}" --audit --output "${KB_LOG_DIR}/security_audit_$(date +%Y%m%d).json"
    
    log "Security audit complete"
}

# Main monitoring function
monitor_knowledge_base() {
    log "Starting knowledge base monitoring..."
    
    # Check if knowledge base exists
    check_kb_exists
    
    # Activate virtual environment
    activate_venv
    
    # Create a backup
    create_backup
    
    # Clean up old backups
    cleanup_old_backups
    
    # Clean up old logs
    cleanup_old_logs

        error "Database integrity check failed, skipping optimization"
    else
        # Optimize database
        optimize_database
    fi
    
    # Run security audit
    run_security_audit

}

This guide explains how to synchronize your knowledge base with a GitHub repository for version control, collaboration, and backup purposes, while maintaining security and preventing data leakage.

## Setup Instructions

### 1. Create a GitHub Repository

First, create a private repository on GitHub to store your knowledge base data:

1. Log in to your GitHub account
2. Click on the "+" icon in the top-right corner and select "New repository"
3. Name your repository (e.g., "knowledge-base")
4. Set visibility to "Private" to ensure your data remains secure
5. Initialize with a README file
6. Click "Create repository"

### 2. Install the Git Sync Script

1. Save the kb_git_sync.sh script to your knowledge base directory:
   
bash
   cp kb_git_sync.sh ~/knowledge_base/bin/


2. Make the script executable:
   
bash
   chmod +x ~/knowledge_base/bin/kb_git_sync.sh


3. Add the script to your path (optional):
   
bash
   echo 'export PATH="$PATH:$HOME/knowledge_base/bin"' >> ~/.bashrc
   source ~/.bashrc


### 3. Configure SSH Access (Recommended)

For secure, password-free access to your GitHub repository:

1. Generate an SSH key if you don't already have one:
   
bash
   ssh-keygen -t ed25519 -C "your_email@example.com"


2. Add the SSH key to your SSH agent:
   
bash
   eval "$(ssh-agent -s)"
   ssh-add ~/.ssh/id_ed25519


3. Add the public key to your GitHub account:
   - Copy the contents of ~/.ssh/id_ed25519.pub
   - Go to GitHub → Settings → SSH and GPG keys → New SSH key
   - Paste the key and give it a descriptive title
   - Click "Add SSH key"

### 4. Initial Sync

Perform your first synchronization with your GitHub repository:

bash
kb_git_sync.sh git@github.com:yourusername/knowledge-base.git


This will:
1. Clone the repository or update it if it already exists
2. Export your knowledge base data in a secure format
3. Commit and push the changes to GitHub

## Usage

### Basic Sync

To synchronize your knowledge base with GitHub (both export and import):

bash
kb_git_sync.sh git@github.com:yourusername/knowledge-base.git


### Export Only

To only export your knowledge base to GitHub without importing:

bash
kb_git_sync.sh git@github.com:yourusername/knowledge-base.git --export-only


### Import Only

To only import from GitHub to your knowledge base without exporting:

bash
kb_git_sync.sh git@github.com:yourusername/knowledge-base.git --import-only


### Force Sync

If there are conflicts or local changes that you want to overwrite:

bash
kb_git_sync.sh git@github.com:yourusername/knowledge-base.git --force


### Skip Security Checks (Not Recommended)

In rare cases, you might need to skip security checks (this is not recommended):

bash
kb_git_sync.sh git@github.com:yourusername/knowledge-base.git --no-security


## Automated Sync

Set up a cron job to automatically synchronize your knowledge base with GitHub:

1. Open your crontab file:
   
bash
   crontab -e


2. Add a line to run the sync script daily (e.g., at 3 AM):
   
0 3 * * * ~/knowledge_base/bin/kb_git_sync.sh git@github.com:yourusername/knowledge-base.git >> ~/knowledge_base/logs/git_sync_cron.log 2>&1


## Security Considerations

The sync script includes several security features:

1. **Sensitive Data Redaction**: Before pushing to GitHub, sensitive data like API keys, passwords, and tokens are automatically redacted.

2. **Private Repository**: Always use a private GitHub repository to store your knowledge base data.

3. **SSH Authentication**: Use SSH keys instead of passwords for more secure authentication.

4. **Pre-Import Backups**: The system automatically creates a backup of your knowledge base before importing data from GitHub.

5. **Selective Export**: You can configure which parts of your knowledge base are synchronized.

## Collaboration Workflow

With your knowledge base synchronized to GitHub, you can collaborate with others:

1. **Add Collaborators**: Go to your GitHub repository → Settings → Collaborators → Add people

2. **Pull Requests**: Collaborators can make changes through pull requests:
   - They fork the repository
   - Make changes in their fork
   - Create a pull request
   - You review and merge the changes

3. **Import Collaborator Changes**: After merging pull requests, run:
   
bash
   kb_git_sync.sh git@github.com:yourusername/knowledge-base.git --import-only


4. **Comment and Review**: Use GitHub's commenting and review features for discussions about knowledge base content.

## Troubleshooting

### Sync Fails with Authentication Error

If you see an error like "Permission denied (publickey)":

1. Ensure your SSH key is added to your GitHub account
2. Check if your SSH agent is running and has the key loaded:
   
bash
   ssh-add -l

3. Try using HTTPS instead (you'll need to enter your GitHub username and password):
   
bash
   kb_git_sync.sh https://github.com/yourusername/knowledge-base.git


### Merge Conflicts

If you encounter merge conflicts:

1. Manually resolve the conflicts:
   
bash
   cd ~/knowledge_base/repo_sync
   git status  # See which files have conflicts
   # Edit the conflicted files
   git add .
   git commit -m "Resolve merge conflicts"
   git push origin main


2. Then run the sync script with the --import-only flag:
   
bash
   kb_git_sync.sh git@github.com:yourusername/knowledge-base.git --import-only


### Data Not Updating

If your knowledge base isn't updating with changes from GitHub:

1. Check the sync logs:
   
bash
   cat ~/knowledge_base/logs/git_sync_*.log


2. Try running the script with verbose output:
   
bash
   bash -x kb_git_sync.sh git@github.com:yourusername/knowledge-base.git


## Best Practices

1. **Regular Sync**: Synchronize regularly to keep your knowledge base and GitHub repository in sync.

2. **Create Branches**: For significant changes, create branches in your GitHub repository.

3. **Detailed Commit Messages**: Write clear commit messages describing your changes.

4. **Review Before Import**: Review changes in the GitHub repository before importing them.

5. **Monitor Logs**: Regularly check the sync logs for any issues or warnings.
