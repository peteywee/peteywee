#!/usr/bin/env bash
set -euo pipefail

DOTFILES_DIR="$HOME/dotfiles"

echo "Updating APT and installing core packages…"
sudo apt update && sudo apt upgrade -y
sudo apt install -y \
  git curl wget build-essential \
  vim tmux htop \
  zsh fonts-powerline \
  python3-pip \
  terraform \
  nodejs npm \
  # add any other packages you need…

echo "Cloning dotfiles repo (if not already)…"
if [ ! -d "$DOTFILES_DIR" ]; then
  git clone https://github.com/peteywee/dotfiles.git "$DOTFILES_DIR"
fi

echo "Creating symlinks…"
for f in bash/.bashrc bash/.bash_aliases git/.gitconfig vim/.vimrc tmux/.tmux.conf input/.inputrc misc/.dir_colors; do
  ln -sfv "$DOTFILES_DIR/$f" "$HOME/$(basename $f)"
done

echo "Loading .bashrc…"
source "$HOME/.bashrc"

echo "All done! Open a new shell to see your changes."
