#!/usr/bin/env bash
# install.sh — download, unpack & install Color Role System in one go

set -euo pipefail

#─── CONFIG ─────────────────────────────────────────────────────────────────────

# GitHub repo and asset name
GITHUB_REPO="youruser/color-system"  
ASSET_PATTERN="color-system-*.tar.gz"  

# Where to install
INSTALL_DIR="${HOME}/.color_system"  

#─── FUNCTIONS ──────────────────────────────────────────────────────────────────

# 1. Find latest release tarball URL
get_asset_url() {
  curl -sSL "https://api.github.com/repos/${GITHUB_REPO}/releases/latest" \
    | grep browser_download_url \
    | cut -d '"' -f 4 \
    | grep "${ASSET_PATTERN}" \
    | head -n1
}

# 2. Download & unpack
download_and_unpack() {
  local url="$1"
  echo "🔽 Downloading ${url}"
  curl -sSL "$url" | tar xzv
}

# 3. Run the installer
run_hook() {
  echo "⚙️  Running install_hook.sh"
  cd color-system-*  
  bash install_hook.sh "$@"
}

#─── MAIN ───────────────────────────────────────────────────────────────────────

echo "📦  Color Role System installer"
ASSET_URL="$(get_asset_url)"
if [[ -z "$ASSET_URL" ]]; then
  echo "❌ Could not find a release asset. Do you have at least one tag?" >&2
  exit 1
fi

download_and_unpack "$ASSET_URL"
run_hook "$@"
echo "🎉  Done! Reload your shell or run 'source ~/.bashrc'."
