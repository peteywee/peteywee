2. Self-Healing Engine for PostgreSQL
Create a robust error detection and automatic repair system:

bash
#!/usr/bin/env bash
# ~/services/watchdog/postgres_healing.sh
set -euo pipefail

LOG_DIR="$HOME/.logs/postgres"
mkdir -p "$LOG_DIR"
LOG_FILE="$LOG_DIR/$(date +%Y%m%d).log"

log() {
  echo "[$(date '+%Y-%m-%d %H:%M:%S')] $*" | tee -a "$LOG_FILE"
}

# Detect common PostgreSQL errors
check_postgres_errors() {
  log "Checking PostgreSQL status..."
  
  # Check if PostgreSQL is running
  if ! systemctl is-active postgresql &>/dev/null; then
    log "PostgreSQL service is not running. Attempting to start..."
    sudo systemctl start postgresql || repair_postgres_service
  fi
  
  # Check connectivity
  if ! psql -c "SELECT 1;" &>/dev/null; then
    log "PostgreSQL connection failed. Diagnosing issue..."
    diagnose_connection_issue
  fi
  
  # Check disk space
  if ! check_postgres_disk_space; then
    log "Low disk space affecting PostgreSQL. Attempting cleanup..."
    cleanup_postgres_disk_space
  fi
  
  # Check for corrupt indexes
  check_corrupt_indexes
  
  # Check for configuration optimizations
  optimize_postgres_config
}

# Repair PostgreSQL service issues
repair_postgres_service() {
  log "Repairing PostgreSQL service..."
  
  # Check common issues that prevent startup
  # 1. Check socket directory permissions
  sudo mkdir -p /var/run/postgresql
  sudo chmod 2775 /var/run/postgresql
  sudo chown postgres:postgres /var/run/postgresql
  
  # 2. Check data directory
  PG_VERSION="16" # Update this to match your version
  sudo chmod 700 "/var/lib/postgresql/${PG_VERSION}/main"
  sudo chown -R postgres:postgres "/var/lib/postgresql/${PG_VERSION}"
  
  # 3. Try starting the service again
  sudo systemctl restart postgresql
  
  # 4. If still failing, check logs
  if ! systemctl is-active postgresql &>/dev/null; then
    log "PostgreSQL still not starting. Latest log entries:"
    sudo tail -n 20 /var/log/postgresql/postgresql-${PG_VERSION}-main.log >> "$LOG_FILE"
  else
    log "PostgreSQL service successfully started."
  fi
}

# Diagnose connection issues
diagnose_connection_issue() {
  log "Diagnosing PostgreSQL connection issues..."
  
  # Check pg_hba.conf configuration
  PG_VERSION="16" # Update this to match your version
  PG_HBA="/etc/postgresql/${PG_VERSION}/main/pg_hba.conf"
  
  if ! grep -q "local   all             all                                     peer" "$PG_HBA"; then
    log "Missing local peer authentication. Adding configuration..."
    echo "local   all             all                                     peer" | sudo tee -a "$PG_HBA"
    sudo systemctl restart postgresql
  fi
  
  # Check if postgres user exists
  if ! id postgres &>/dev/null; then
    log "PostgreSQL user does not exist. Creating user..."
    sudo useradd -r -m -d /var/lib/postgresql -s /bin/bash postgres
  fi
  
  # Check if user has a database
  if ! sudo -u postgres psql -c "\\du" | grep -q "$USER"; then
    log "User $USER does not exist in PostgreSQL. Creating..."
    sudo -u postgres createuser --superuser "$USER"
    sudo -u postgres createdb "$USER"
  fi
  
  log "Connection issue diagnosis complete."
}

# Check PostgreSQL disk space
check_postgres_disk_space() {
  local threshold=90
  local usage=$(df -h /var/lib/postgresql | tail -n 1 | awk '{print $5}' | tr -d '%')
  
  if (( usage > threshold )); then
    log "WARNING: PostgreSQL data directory disk space critical ($usage% used)"
    return 1
  fi
  
  return 0
}

# Clean up PostgreSQL disk space
cleanup_postgres_disk_space() {
  log "Cleaning up PostgreSQL disk space..."
  
  # 1. Vacuum full databases
  sudo -u postgres psql -c "VACUUM FULL;"
  
  # 2. Remove old logs
  sudo find /var/log/postgresql -type f -name "*.log.*" -mtime +7 -delete
  
  # 3. Clean up pg_wal (transaction logs)
  sudo -u postgres psql -c "SELECT pg_switch_wal();"
  
  log "PostgreSQL disk cleanup complete."
}

# Check for corrupt indexes
check_corrupt_indexes() {
  log "Checking for corrupt indexes..."
  
  # Get list of databases
  databases=$(sudo -u postgres psql -t -c "SELECT datname FROM pg_database WHERE datistemplate = false;")
  
  for db in $databases; do
    log "Checking indexes in database: $db"
    sudo -u postgres psql -d "$db" -c "REINDEX DATABASE $db;"
  done
  
  log "Index check complete."
}

# Optimize PostgreSQL configuration
optimize_postgres_config() {
  PG_VERSION="16" # Update this to match your version
  PG_CONF="/etc/postgresql/${PG_VERSION}/main/postgresql.conf"
  
  # Get system memory in MB
  mem_kb=$(grep MemTotal /proc/meminfo | awk '{print $2}')
  mem_mb=$((mem_kb / 1024))
  
  # Calculate optimal shared_buffers (25% of RAM)
  shared_buffers=$((mem_mb / 4))
  
  # Calculate work_mem (1/16 of RAM)
  work_mem=$((mem_mb / 16))
  
  log "Optimizing PostgreSQL configuration based on system with ${mem_mb}MB RAM..."
  
  # Only update if values are significantly different
  current_shared_buffers=$(sudo grep "^shared_buffers" "$PG_CONF" | awk '{print $3}' | sed 's/MB//')
  
  if [[ -z "$current_shared_buffers" ]] || (( current_shared_buffers < shared_buffers / 2 )); then
    log "Optimizing shared_buffers to ${shared_buffers}MB"
    sudo sed -i "s/^shared_buffers.*/shared_buffers = ${shared_buffers}MB/" "$PG_CONF"
    optimization_needed=true
  fi
  
  current_work_mem=$(sudo grep "^work_mem" "$PG_CONF" | awk '{print $3}' | sed 's/MB//')
  
  if [[ -z "$current_work_mem" ]] || (( current_work_mem < work_mem / 2 )); then
    log "Optimizing work_mem to ${work_mem}MB"
    sudo sed -i "s/^work_mem.*/work_mem = ${work_mem}MB/" "$PG_CONF"
    optimization_needed=true
  fi
  
  # Restart if optimizations were made
  if [[ "$optimization_needed" = true ]]; then
    log "Configuration optimized, restarting PostgreSQL..."
    sudo systemctl restart postgresql
  else
    log "PostgreSQL configuration is already optimal."
  fi
}

#!/usr/bin/env bash
set -euo pipefail

BACKUP_DIR="$HOME/backups/postgres"
LOG_DIR="$HOME/.logs/postgres"
mkdir -p "$BACKUP_DIR" "$LOG_DIR"
LOG_FILE="$LOG_DIR/backup_$(date +%Y%m%d).log"

log() {

#!/usr/bin/env bash
set -euo pipefail

LOG_DIR="$HOME/.logs/postgres"
mkdir -p "$LOG_DIR"
LOG_FILE="$LOG_DIR/restore_$(date +%Y%m%d).log"

log() {
  echo "[$(date '+%Y-%m-%d %H:%M:%S')] $*" | tee -a "$LOG_FILE"
}

if [ $# -lt 1 ]; then
  echo "Usage: $0 <backup_file> [database_name]"
  echo "  If database_name is not provided, it will be extracted from the backup filename."
  exit 1
fi

BACKUP_FILE="$1"
DB_NAME="${2:-}"

if [ ! -f "$BACKUP_FILE" ]; then
  log "Error: Backup file not found: $BACKUP_FILE"
  exit 1
fi

# Extract database name from backup file if not provided
if [ -z "$DB_NAME" ]; then
  filename=$(basename "$BACKUP_FILE")
  DB_NAME=$(echo "$filename" | sed -E 's/^([^_]+)_.*/\1/')
  log "Extracted database name from filename: $DB_NAME"
fi

log "Restoring $BACKUP_FILE to database $DB_NAME..."

# Check if database exists
if psql -lqt | cut -d \| -f 1 | grep -qw "$DB_NAME"; then
  read -p "Database $DB_NAME already exists. Drop and recreate? (y/n): " confirm
  if [[ "$confirm" == [yY] ]]; then

    psql -c "DROP DATABASE IF EXISTS \"$DB_NAME\";"

    log "Restoration canceled by user"
    exit 0
  fi
fi

# Create the database

psql -c "CREATE DATABASE \"$DB_NAME\";"

# Restore the database
log "Restoring data from $BACKUP_FILE"
pg_restore -d "$DB_NAME" -v "$BACKUP_FILE" 2>&1 | tee -a "$LOG_FILE"

if [ ${PIPESTATUS[0]} -eq 0 ]; then

else

  log "The database may still be usable."
fi

log "Analyzing database to update statistics"
psql -c "ANALYZE;" "$DB_NAME"

log "Restoration process complete for database: $DB_NAME"
EOF

  psql -c "ALTER DATABASE \"$DB_NAME\" OWNER TO \"$OWNER\";"

# Install extensions if specified
if [ -n "$EXTENSIONS" ]; then

  IFS=',' read -ra EXT_ARRAY <<< "$EXTENSIONS"

#!/usr/bin/env bash
# ~/dotfiles/postgres/scripts/pg_healing.sh
set -euo pipefail

LOG_DIR="$HOME/.logs/postgres"
mkdir -p "$LOG_DIR"
LOG_FILE="$LOG_DIR/healing_$(date +%Y%m%d).log"

log() {
  echo "[$(date '+%Y-%m-%d %H:%M:%S')] $*" | tee -a "$LOG_FILE"
}

# Get PostgreSQL version
PG_VERSION=$(pg_config --version | awk '{print $2}' | cut -d. -f1)
PG_DATA_DIR="/var/lib/postgresql/${PG_VERSION}/main"
PG_CONF="/etc/postgresql/${PG_VERSION}/main/postgresql.conf"
PG_HBA="/etc/postgresql/${PG_VERSION}/main/pg_hba.conf"

# Detect common PostgreSQL errors
check_postgres_errors() {
  log "Checking PostgreSQL status..."
  
  # Check if PostgreSQL is running
  if ! systemctl is-active postgresql &>/dev/null; then
    log "PostgreSQL service is not running. Attempting to start..."
    sudo systemctl start postgresql || repair_postgres_service
  fi
  
  # Check connectivity
  if ! psql -U postgres -c "SELECT 1;" &>/dev/null; then
    log "PostgreSQL connection failed. Diagnosing issue..."
    diagnose_connection_issue
  fi
  
  # Check disk space
  if ! check_postgres_disk_space; then
    log "Low disk space affecting PostgreSQL. Attempting cleanup..."
    cleanup_postgres_disk_space
  fi
  
  # Check for corrupt indexes
  check_corrupt_indexes
  
  # Check for configuration optimizations
  optimize_postgres_config
  
  # Check for pending maintenance tasks
  check_maintenance_tasks
  
  # Check for autovacuum settings
  check_autovacuum_settings
  
  log "PostgreSQL health check completed."
}

# Repair PostgreSQL service issues
repair_postgres_service() {
  log "Repairing PostgreSQL service..."
  
  # Check common issues that prevent startup
  # 1. Check socket directory permissions
  sudo mkdir -p /var/run/postgresql
  sudo chmod 2775 /var/run/postgresql
  sudo chown postgres:postgres /var/run/postgresql
  
  # 2. Check data directory
  sudo chmod 700 "$PG_DATA_DIR"
  sudo chown -R postgres:postgres "/var/lib/postgresql/${PG_VERSION}"
  
  # 3. Check for running postmaster process that might prevent startup
  if [ -f "$PG_DATA_DIR/postmaster.pid" ]; then
    log "postmaster.pid exists, checking if process is still running"
    PID=$(head -1 "$PG_DATA_DIR/postmaster.pid")
    if ! ps -p "$PID" > /dev/null; then
      log "Stale PID file found, removing"
      sudo rm -f "$PG_DATA_DIR/postmaster.pid"
    fi
  fi
  
  # 4. Try starting the service again
  sudo systemctl restart postgresql
  
  # 5. If still failing, check logs
  if ! systemctl is-active postgresql &>/dev/null; then
    log "PostgreSQL still not starting. Latest log entries:"
    sudo tail -n 20 /var/log/postgresql/postgresql-${PG_VERSION}-main.log >> "$LOG_FILE"
    
    # Check for disk full condition
    if grep -q "No space left on device" /var/log/postgresql/postgresql-${PG_VERSION}-main.log; then
      log "Disk full condition detected, emergency cleanup"
      emergency_cleanup
    fi
  else
    log "PostgreSQL service successfully started."
  fi
}

# Emergency cleanup when disk is full
emergency_cleanup() {
  log "Performing emergency cleanup to free disk space..."
  
  # Clean up temporary files
  sudo find /tmp -type f -mtime +1 -delete
  
  # Clean up journal logs
  sudo journalctl --vacuum-time=1d
  
  # Clean up old PostgreSQL logs
  sudo find /var/log/postgresql -type f -name "*.log.*" -delete
  
  # Try starting PostgreSQL again
  sudo systemctl restart postgresql
}

# Diagnose connection issues
diagnose_connection_issue() {
  log "Diagnosing PostgreSQL connection issues..."
  
  # Check pg_hba.conf configuration
  if ! grep -q "local.*all.*all.*peer" "$PG_HBA"; then
    log "Missing local peer authentication. Adding configuration..."
    echo "local   all             all                                     peer" | sudo tee -a "$PG_HBA"
    sudo systemctl restart postgresql
  fi
  
  # Check if postgres user exists
  if ! id postgres &>/dev/null; then
    log "PostgreSQL user does not exist. Creating user..."
    sudo useradd -r -m -d /var/lib/postgresql -s /bin/bash postgres
  fi
  
  # Check if user has a database
  if ! sudo -u postgres psql -c "\\du" | grep -q "$USER"; then
    log "User $USER does not exist in PostgreSQL. Creating..."
    sudo -u postgres createuser --superuser "$USER"
    sudo -u postgres createdb "$USER"
  fi
  
  # Check for authentication method issues
  if grep -q "local.*all.*all.*trust" "$PG_HBA"; then
    log "Warning: 'trust' authentication method found, this is insecure"
  fi
  
  log "Connection issue diagnosis complete."
}

# Check PostgreSQL disk space
check_postgres_disk_space() {
  local threshold=90
  local usage=$(df -h /var/lib/postgresql | tail -n 1 | awk '{print $5}' | tr -d '%')
  
  if (( usage > threshold )); then
    log "WARNING: PostgreSQL data directory disk space critical ($usage% used)"
    return 1
  fi
  
  return 0
}

# Clean up PostgreSQL disk space
cleanup_postgres_disk_space() {
  log "Cleaning up PostgreSQL disk space..."
  
  # 1. Vacuum full databases
  log "Running VACUUM FULL on all databases"
  for db in $(sudo -u postgres psql -t -c "SELECT datname FROM pg_database WHERE datistemplate = false;"); do
    db=$(echo "$db" | xargs)  # Trim whitespace
    log "Vacuuming database: $db"
    sudo -u postgres psql -d "$db" -c "VACUUM FULL;"
  done
  
  # 2. Remove old logs
  log "Removing old PostgreSQL logs"
  sudo find /var/log/postgresql -type f -name "*.log.*" -mtime +7 -delete
  
  # 3. Clean up pg_wal (transaction logs)
  log "Switching WAL files"
  sudo -u postgres psql -c "SELECT pg_switch_wal();"
  
  # 4. Check bloat and reindex if necessary
  log "Checking for table bloat"
  for db in $(sudo -u postgres psql -t -c "SELECT datname FROM pg_database WHERE datistemplate = false;"); do
    db=$(echo "$db" | xargs)  # Trim whitespace
    log "Checking database: $db for bloated tables"
    
    # Find bloated tables (more than 50% bloat)
    bloated_tables=$(sudo -u postgres psql -d "$db" -t -c "
      SELECT schemaname || '.' || tablename
      FROM pg_tables
      JOIN pg_class ON tablename = relname
      JOIN pg_namespace ON schemaname = nspname
      WHERE pg_total_relation_size(schemaname || '.' || tablename) > 10 * 1024 * 1024  -- Only check tables larger than 10MB
      AND pg_total_relation_size(schemaname || '.' || tablename) * 0.5 < pg_total_relation_size(schemaname || '.' || tablename) - pg_relation_size(schemaname || '.' || tablename);
    ")
    
    if [ -n "$bloated_tables" ]; then
      for table in $bloated_tables; do
        log "Rebuilding bloated table: $table"
        sudo -u postgres psql -d "$db" -c "VACUUM FULL $table;"
      done
    else
      log "No significantly bloated tables found in $db"
    fi
  done
  
  log "PostgreSQL disk cleanup complete."
}

# Check for corrupt indexes
check_corrupt_indexes() {
  log "Checking for corrupt indexes..."
  
  # Get list of databases
  databases=$(sudo -u postgres psql -t -c "SELECT datname FROM pg_database WHERE datistemplate = false;")
  
  for db in $databases; do
    db=$(echo "$db" | xargs)  # Trim whitespace
    log "Checking indexes in database: $db"
    
    # Check for any issues with database consistency
    if sudo -u postgres pg_amcheck -d "$db" --install-missing 2>&1 | grep -i -E 'error|corrupt'; then
      log "Corruption detected in database: $db, reindexing"
      sudo -u postgres psql -d "$db" -c "REINDEX DATABASE $db;"
    else
      log "No index corruption detected in database: $db"
    fi
  done
  
  log "Index check complete."
}

# Optimize PostgreSQL configuration
optimize_postgres_config() {
  # Get system memory in MB
  mem_kb=$(grep MemTotal /proc/meminfo | awk '{print $2}')
  mem_mb=$((mem_kb / 1024))
  
  # Calculate optimal settings based on available memory
  shared_buffers=$((mem_mb / 4))
  effective_cache_size=$((mem_mb / 2))
  work_mem=$((mem_mb / 16))
  maintenance_work_mem=$((mem_mb / 16))
  
  log "Checking PostgreSQL configuration optimization (System RAM: ${mem_mb}MB)..."
  
  # Check current settings
  current_shared_buffers=$(sudo grep "^shared_buffers" "$PG_CONF" | awk '{print $3}' | sed 's/MB//')
  current_effective_cache_size=$(sudo grep "^effective_cache_size" "$PG_CONF" | awk '{print $3}' | sed 's/MB//')
  current_work_mem=$(sudo grep "^work_mem" "$PG_CONF" | awk '{print $3}' | sed 's/MB//')
  current_maintenance_work_mem=$(sudo grep "^maintenance_work_mem" "$PG_CONF" | awk '{print $3}' | sed 's/MB//')
  
  optimization_needed=false
  
  # Check if shared_buffers needs optimization
  if [[ -z "$current_shared_buffers" ]] || (( current_shared_buffers < shared_buffers / 2 )); then
    log "Optimizing shared_buffers to ${shared_buffers}MB (was: ${current_shared_buffers:-not set})"
    sudo sed -i "/^shared_buffers/c\\shared_buffers = ${shared_buffers}MB" "$PG_CONF"
    optimization_needed=true
  fi
  
  # Check if effective_cache_size needs optimization
  if [[ -z "$current_effective_cache_size" ]] || (( current_effective_cache_size < effective_cache_size / 2 )); then
    log "Optimizing effective_cache_size to ${effective_cache_size}MB (was: ${current_effective_cache_size:-not set})"
    if grep -q "^effective_cache_size" "$PG_CONF"; then
      sudo sed -i "/^effective_cache_size/c\\effective_cache_size = ${effective_cache_size}MB" "$PG_CONF"
    else
      echo "effective_cache_size = ${effective_cache_size}MB" | sudo tee -a "$PG_CONF" > /dev/null
    fi
    optimization_needed=true
  fi
  
  # Check if work_mem needs optimization
  if [[ -z "$current_work_mem" ]] || (( current_work_mem < work_mem / 2 )); then
    log "Optimizing work_mem to ${work_mem}MB (was: ${current_work_mem:-not set})"
    if grep -q "^work_mem" "$PG_CONF"; then
      sudo sed -i "/^work_mem/c\\work_mem = ${work_mem}MB" "$PG_CONF"
    else
      echo "work_mem = ${work_mem}MB" | sudo tee -a "$PG_CONF" > /dev/null
    fi
    optimization_needed=true
  fi
  
  # Check if maintenance_work_mem needs optimization
  if [[ -z "$current_maintenance_work_mem" ]] || (( current_maintenance_work_mem < maintenance_work_mem / 2 )); then
    log "Optimizing maintenance_work_mem to ${maintenance_work_mem}MB (was: ${current_maintenance_work_mem:-not set})"
    if grep -q "^maintenance_work_mem" "$PG_CONF"; then
      sudo sed -i "/^maintenance_work_mem/c\\maintenance_work_mem = ${maintenance_work_mem}MB" "$PG_CONF"
    else
      echo "maintenance_work_mem = ${maintenance_work_mem}MB" | sudo tee -a "$PG_CONF" > /dev/null
    fi
    optimization_needed=true
  fi
  
  # Restart if optimizations were made
  if [[ "$optimization_needed" = true ]]; then
    log "Configuration optimized, restarting PostgreSQL..."
    sudo systemctl restart postgresql
  else
    log "PostgreSQL configuration is already optimal."
  fi
}

# Check for pending maintenance tasks
check_maintenance_tasks() {
  log "Checking for pending maintenance tasks..."
  
  # Get list of databases
  databases=$(sudo -u postgres psql -t -c "SELECT datname FROM pg_database WHERE datistemplate = false;")
  
  for db in $databases; do
    db=$(echo "$db" | xargs)  # Trim whitespace
    log "Checking maintenance tasks for database: $db"
    
    # Check for tables that haven't been analyzed recently
    unanalyzed_tables=$(sudo -u postgres psql -d "$db" -t -c "
      SELECT schemaname || '.' || relname
      FROM pg_stat_user_tables
      WHERE n_live_tup > 10000 AND
            (last_analyze IS NULL OR last_analyze < NOW() - INTERVAL '1 day')
      LIMIT 5;
    ")
    
    if [ -n "$unanalyzed_tables" ]; then
      log "Running ANALYZE on tables that need statistics updates:"
      for table in $unanalyzed_tables; do
        log "  - Analyzing table: $table"
        sudo -u postgres psql -d "$db" -c "ANALYZE VERBOSE $table;"
      done
    else
      log "All tables in $db have recent statistics."
    fi
    
    # Check for tables that need vacuuming
    need_vacuum_tables=$(sudo -u postgres psql -d "$db" -t -c "
      SELECT schemaname || '.' || relname,
             n_dead_tup,
             round(n_dead_tup * 100.0 / greatest(n_live_tup, 1), 1) as dead_ratio
      FROM pg_stat_user_tables
      WHERE n_dead_tup > 1000 OR (n_live_tup > 10000 AND n_dead_tup > n_live_tup * 0.05)
      ORDER BY n_dead_tup DESC
      LIMIT 5;
    ")
    
    if [ -n "$need_vacuum_tables" ]; then
      log "Running VACUUM on tables with high dead tuple counts:"
      echo "$need_vacuum_tables" | while read -r line; do
        table=$(echo "$line" | awk -F'|' '{print $1}' | xargs)
        dead_tuples=$(echo "$line" | awk -F'|' '{print $2}' | xargs)
        dead_ratio=$(echo "$line" | awk -F'|' '{print $3}' | xargs)
        
        log "  - Vacuuming table: $table (dead tuples: $dead_tuples, ratio: ${dead_ratio}%)"
        sudo -u postgres psql -d "$db" -c "VACUUM VERBOSE $table;"
      done
    else
      log "No tables in $db need immediate vacuuming."
    fi
  done
  
  log "Maintenance tasks check complete."
}

# Check autovacuum settings
check_autovacuum_settings() {
  log "Checking autovacuum settings..."
  
  # Check if autovacuum is enabled
  autovacuum_enabled=$(sudo -u postgres psql -t -c "SHOW autovacuum;")
  
  if [ "${autovacuum_enabled}" != " on" ]; then
    log "Autovacuum is disabled! Enabling..."
    sudo -u postgres psql -c "ALTER SYSTEM SET autovacuum = on;"
    sudo -u postgres psql -c "SELECT pg_reload_conf();"
  else
    log "Autovacuum is enabled."
  fi
  
  # Check current settings
  autovacuum_max_workers=$(sudo -u postgres psql -t -c "SHOW autovacuum_max_workers;" | xargs)
  autovacuum_naptime=$(sudo -u postgres psql -t -c "SHOW autovacuum_naptime;" | xargs)
  autovacuum_vacuum_scale_factor=$(sudo -u postgres psql -t -c "SHOW autovacuum_vacuum_scale_factor;" | xargs)
  autovacuum_analyze_scale_factor=$(sudo -u postgres psql -t -c "SHOW autovacuum_analyze_scale_factor;" | xargs)
  
  # Get system specs for optimization
  cpu_cores=$(grep -c ^processor /proc/cpuinfo)
  mem_gb=$((mem_mb / 1024))
  
  # Log current settings
  log "Current autovacuum settings:"
  log "  - autovacuum_max_workers: $autovacuum_max_workers"
  log "  - autovacuum_naptime: $autovacuum_naptime"
  log "  - autovacuum_vacuum_scale_factor: $autovacuum_vacuum_scale_factor"
  log "  - autovacuum_analyze_scale_factor: $autovacuum_analyze_scale_factor"
  
  # Recommendations based on system specs
  recommended_workers=$(( cpu_cores / 4 + 1 ))
  [[ $recommended_workers -lt 3 ]] && recommended_workers=3
  [[ $recommended_workers -gt 8 ]] && recommended_workers=8
  
  # For busy systems with many tables, use more aggressive settings
  if [[ $autovacuum_max_workers -lt $recommended_workers ]]; then
    log "Recommending autovacuum_max_workers increase from $autovacuum_max_workers to $recommended_workers"
    sudo -u postgres psql -c "ALTER SYSTEM SET autovacuum_max_workers = $recommended_workers;"
    settings_changed=true
  fi
  
  if [[ "${autovacuum_vacuum_scale_factor}" != "0.01" ]]; then
    log "Recommending more aggressive autovacuum_vacuum_scale_factor: 0.01 (was: $autovacuum_vacuum_scale_factor)"
    sudo -u postgres psql -c "ALTER SYSTEM SET autovacuum_vacuum_scale_factor = 0.01;"
    settings_changed=true
  fi
  
  if [[ "${autovacuum_analyze_scale_factor}" != "0.01" ]]; then
    log "Recommending more aggressive autovacuum_analyze_scale_factor: 0.01 (was: $autovacuum_analyze_scale_factor)"
    sudo -u postgres psql -c "ALTER SYSTEM SET autovacuum_analyze_scale_factor = 0.01;"
    settings_changed=true
  fi
  
  if [[ "$settings_changed" = true ]]; then
    log "Autovacuum settings updated, reloading configuration..."
    sudo -u postgres psql -c "SELECT pg_reload_conf();"
  else
    log "Autovacuum settings are optimal."
  fi
}

# Create update config script
cat > "$SCRIPTS_DIR/update-config.sh" << 'EOF'
#!/bin/bash
set -e

# Update PostgreSQL configuration with custom settings
echo "include_if_exists = '/etc/postgresql/custom/postgresql.custom.conf'" >> "$PGDATA/postgresql.conf"

# Set memory parameters based on container limits
if [ -f /sys/fs/cgroup/memory/memory.limit_in_bytes ]; then
  mem_limit_bytes=$(cat /sys/fs/cgroup/memory/memory.limit_in_bytes)
  
  # Convert to MB with a 10% safety margin
  mem_limit_mb=$(( mem_limit_bytes / 1024 / 1024 * 9 / 10 ))
  
  if [ "$mem_limit_mb" -gt 0 ]; then
    echo "Container memory limit detected: ${mem_limit_mb}MB"
    
    # Calculate optimal settings
    shared_buffers=$(( mem_limit_mb / 4 ))
    effective_cache_size=$(( mem_limit_mb / 2 ))
    work_mem=$(( mem_limit_mb / 16 ))
    maintenance_work_mem=$(( mem_limit_mb / 8 ))
    
    # Ensure minimums
    [ "$shared_buffers" -lt 128 ] && shared_buffers=128
    [ "$effective_cache_size" -lt 256 ] && effective_cache_size=256
    [ "$work_mem" -lt 4 ] && work_mem=4
    [ "$maintenance_work_mem" -lt 16 ] && maintenance_work_mem=16
    
    # Update configuration
    sed -i "s/^shared_buffers = .*/shared_buffers = ${shared_buffers}MB/" /etc/postgresql/custom/postgresql.custom.conf
    sed -i "s/^effective_cache_size = .*/effective_cache_size = ${effective_cache_size}MB/" /etc/postgresql/custom/postgresql.custom.conf
    sed -i "s/^work_mem = .*/work_mem = ${work_mem}MB/" /etc/postgresql/custom/postgresql.custom.conf
    sed -i "s/^maintenance_work_mem = .*/maintenance_work_mem = ${maintenance_work_mem}MB/" /etc/postgresql/custom/postgresql.custom.conf
    
    echo "PostgreSQL memory settings optimized for container"
  fi
fi

echo "PostgreSQL custom configuration applied"
EOF
chmod +x "$SCRIPTS_DIR/update-config.sh"

install_extensions() {
  DB_NAME="$1"

  psql -v ON_ERROR_STOP=1 --username "$POSTGRES_USER" --dbname "$DB_NAME" <<-EOSQL

    CREATE EXTENSION IF NOT EXISTS "adminpack";
    CREATE EXTENSION IF NOT EXISTS "pgcrypto";
    CREATE EXTENSION IF NOT EXISTS "unaccent";
    CREATE EXTENSION IF NOT EXISTS "pg_trgm";
EOSQL
  
  # PostGIS extension (may fail if not available, which is fine)
  psql -v ON_ERROR_STOP=0 --username "$POSTGRES_USER" --dbname "$DB_NAME" <<-EOSQL
    CREATE EXTENSION IF NOT EXISTS "postgis";
    CREATE EXTENSION IF NOT EXISTS "postgis_topology";
EOSQL
  
  # More specialized extensions - try with error handling
  for ext in "pgaudit" "pg_cron" "pg_partman" "pg_repack" "pglogical" "pgvector"; do

      echo "Extension $ext installed successfully"
    else

    fi
  done

# Create backup script for Docker PostgreSQL
cat > "$DOCKER_DIR/backup-postgres.sh" << 'EOF'
#!/usr/bin/env bash
set -euo pipefail

DOCKER_DIR="$HOME/dotfiles/docker/postgres"
BACKUP_DIR="$DOCKER_DIR/backups"
CONTAINER_NAME="postgres_db"
LOG_FILE="$BACKUP_DIR/backup_$(date +%Y%m%d).log"

mkdir -p "$BACKUP_DIR"

log() {
  echo "[$(date '+%Y-%m-%d %H:%M:%S')] $*" | tee -a "$LOG_FILE"
}

# Get date for backup file
DATE=$(date +%Y%m%d_%H%M%S)
log "Starting PostgreSQL Docker backup - $DATE"

# Check if container is running
if ! docker ps | grep -q "$CONTAINER_NAME"; then
  log "ERROR: PostgreSQL container '$CONTAINER_NAME' is not running."
  exit 1
fi

# Create year/month directory structure
YEAR_MONTH=$(date +%Y/%m)
mkdir -p "$BACKUP_DIR/$YEAR_MONTH"

# Get list of databases excluding template databases
databases=$(docker exec "$CONTAINER_NAME" psql -U postgres -t -c "SELECT datname FROM pg_database WHERE datistemplate = false AND datname != 'postgres';")

# Backup each database
for db in $databases; do
  db=$(echo "$db" | xargs)  # Trim whitespace
  log "Backing up database: $db"
  
  backup_file="$BACKUP_DIR/$YEAR_MONTH/${db}_${DATE}.dump"
  
  # Create backup with compression
  docker exec "$CONTAINER_NAME" pg_dump -U postgres -Fc -Z9 "$db" > "$backup_file"
  
  if [ $? -eq 0 ]; then
    log "Backup successful: $backup_file ($(du -h "$backup_file" | cut -f1))"
    
    # Create a symlink to the latest backup
    ln -sf "$backup_file" "$BACKUP_DIR/${db}_latest.dump"
  else
    log "ERROR: Backup failed for database $db"
  fi
done

# Backup postgres database with globals (roles, tablespaces)
log "Backing up globals and postgres database"
docker exec "$CONTAINER_NAME" pg_dumpall -U postgres -g > "$BACKUP_DIR/$YEAR_MONTH/globals_${DATE}.sql"
docker exec "$CONTAINER_NAME" pg_dump -U postgres -Fc -Z9 postgres > "$BACKUP_DIR/$YEAR_MONTH/postgres_${DATE}.dump"

# Clean up old backups (keep last 30 days, monthly backups for a year)
log "Cleaning up old backups..."

# Keep daily backups for 30 days
find "$BACKUP_DIR" -type f -name "*.dump" -mtime +30 -not -name "*_01_*.dump" -delete

# Keep monthly backups for 1 year
find "$BACKUP_DIR" -type f -name "*_01_*.dump" -mtime +365 -delete

log "PostgreSQL Docker backups completed. See $BACKUP_DIR for all backups."
EOF
chmod +x "$DOCKER_DIR/backup-postgres.sh"

# Create restore script for Docker PostgreSQL
cat > "$DOCKER_DIR/restore-postgres.sh" << 'EOF'
#!/usr/bin/env bash
set -euo pipefail

DOCKER_DIR="$HOME/dotfiles/docker/postgres"
BACKUP_DIR="$DOCKER_DIR/backups"
CONTAINER_NAME="postgres_db"
LOG_FILE="$BACKUP_DIR/restore_$(date +%Y%m%d).log"

log() {
  echo "[$(date '+%Y-%m-%d %H:%M:%S')] $*" | tee -a "$LOG_FILE"
}

if [ $# -lt 1 ]; then
  echo "Usage: $0 <backup_file> [database_name]"
  echo "  If database_name is not provided, it will be extracted from the backup filename."
  exit 1
fi

BACKUP_FILE="$1"
DB_NAME="${2:-}"

if [ ! -f "$BACKUP_FILE" ]; then
  log "Error: Backup file not found: $BACKUP_FILE"
  exit 1
fi

# Check if container is running
if ! docker ps | grep -q "$CONTAINER_NAME"; then
  log "ERROR: PostgreSQL container '$CONTAINER_NAME' is not running."
  exit 1
fi

# Extract database name from backup file if not provided
if [ -z "$DB_NAME" ]; then
  filename=$(basename "$BACKUP_FILE")
  DB_NAME=$(echo "$filename" | sed -E 's/^([^_]+)_.*/\1/')
  log "Extracted database name from filename: $DB_NAME"
fi

log "Restoring $BACKUP_FILE to database $DB_NAME..."

# Check if the backup is a globals file
if [[ "$BACKUP_FILE" == *"globals"* ]]; then
  log "Restoring global objects (roles, tablespaces)..."
  cat "$BACKUP_FILE" | docker exec -i "$CONTAINER_NAME" psql -U postgres
  log "Global objects restored."
  exit 0
fi

# Check if database exists
db_exists=$(docker exec "$CONTAINER_NAME" psql -U postgres -tAc "SELECT 1 FROM pg_database WHERE datname='$DB_NAME'")

if [ "$db_exists" = "1" ]; then
  read -p "Database $DB_NAME already exists. Drop and recreate? (y/n): " confirm
  if [[ "$confirm" == [yY] ]]; then
    log "Dropping existing database $DB_NAME"
    docker exec "$CONTAINER_NAME" psql -U postgres -c "DROP DATABASE IF EXISTS \"$DB_NAME\";"
  else
    log "Restoration canceled by user"
    exit 0
  fi
fi

# Create the database
log "Creating database $DB_NAME"
docker exec "$CONTAINER_NAME" psql -U postgres -c "CREATE DATABASE \"$DB_NAME\";"

# Restore the database
log "Restoring data from $BACKUP_FILE"
cat "$BACKUP_FILE" | docker exec -i "$CONTAINER_NAME" pg_restore -U postgres -d "$DB_NAME" -v

if [ ${PIPESTATUS[1]} -eq 0 ]; then
  log "Database restored successfully!"
else
  log "Warning: Some errors occurred during restoration. Check the log for details."
  log "The database may still be usable."
fi

log "Analyzing database to update statistics"
docker exec "$CONTAINER_NAME" psql -U postgres -c "ANALYZE;" "$DB_NAME"

log "Restoration process complete for database: $DB_NAME"
EOF
chmod +x "$DOCKER_DIR/restore-postgres.sh"

#!/usr/bin/env bash
set -euo pipefail

LOG_DIR="$HOME/.logs/postgres"
mkdir -p "$LOG_DIR"
LOG_FILE="$LOG_DIR/maintenance_$(date +%Y%m%d).log"

log() {
  echo "[$(date '+%Y-%m-%d %H:%M:%S')] $*" | tee -a "$LOG_FILE"
}

# Determine maintenance type based on day of week/month
DOW=$(date +%u)  # 1-7 (Monday-Sunday)
DOM=$(date +%d)  # 01-31

# Daily maintenance (always runs)
log "Running daily maintenance tasks..."
sudo -u postgres psql -c "SELECT pg_stat_statements_reset();"

# Weekly maintenance (Sundays)
if [ "$DOW" -eq 7 ]; then
  log "Running weekly maintenance tasks..."
  
  # Get all databases
  databases=$(sudo -u postgres psql -t -c "SELECT datname FROM pg_database WHERE datistemplate = false;")
  
  for db in $databases; do
    db=$(echo "$db" | xargs)  # Trim whitespace
    log "Running weekly maintenance on database: $db"
    
    # Analyze all tables
    sudo -u postgres psql -d "$db" -c "ANALYZE VERBOSE;"
    
    # Check for unused indexes
    unused_indexes=$(sudo -u postgres psql -d "$db" -t -c "
      SELECT indexrelname, relname
      FROM pg_stat_user_indexes
      JOIN pg_index USING (indexrelid)
      WHERE idx_scan = 0 AND indisunique IS FALSE;
    ")
    
    if [ -n "$unused_indexes" ]; then
      log "Unused indexes found in $db:"
      echo "$unused_indexes" | while read -r line; do
        log "  - $line"
      done
    fi
  done
fi

# Monthly maintenance (1st day of month)
if [ "$DOM" -eq "01" ]; then
  log "Running monthly maintenance tasks..."
  
  # Get all databases
  databases=$(sudo -u postgres psql -t -c "SELECT datname FROM pg_database WHERE datistemplate = false;")
  
  for db in $databases; do
    db=$(echo "$db" | xargs)  # Trim whitespace
    log "Running monthly maintenance on database: $db"
    
    # Find and vacuum bloated tables
    bloated_tables=$(sudo -u postgres psql -d "$db" -t -c "
      SELECT schemaname || '.' || tablename,
             pg_size_pretty(pg_total_relation_size(schemaname || '.' || tablename)) as size
      FROM pg_tables
      JOIN pg_class ON tablename = relname
      JOIN pg_namespace ON schemaname = nspname
      WHERE pg_total_relation_size(schemaname || '.' || tablename) > 100 * 1024 * 1024  -- Tables larger than 100MB
      AND pg_total_relation_size(schemaname || '.' || tablename) * 0.3 < pg_total_relation_size(schemaname || '.' || tablename) - pg_relation_size(schemaname || '.' || tablename)
      ORDER BY pg_total_relation_size(schemaname || '.' || tablename) DESC;
    ")
    
    if [ -n "$bloated_tables" ]; then
      log "Bloated tables found in $db:"
      echo "$bloated_tables" | while read -r line; do
        table=$(echo "$line" | awk -F'|' '{print $1}' | xargs)
        size=$(echo "$line" | awk -F'|' '{print $2}' | xargs)
        
        log "  - Vacuuming bloated table: $table ($size)"
        sudo -u postgres psql -d "$db" -c "VACUUM FULL VERBOSE $table;"
      done
    else
      log "No significantly bloated tables found in $db"
    fi
    
    # Database-wide statistics
    log "Database statistics for $db:"
    sudo -u postgres psql -d "$db" -c "
      SELECT 
        pg_size_pretty(pg_database_size('$db')) as database_size,
        (SELECT COUNT(*) FROM pg_stat_activity WHERE datname = '$db') as active_connections,
        (SELECT COUNT(*) FROM pg_tables WHERE schemaname = 'public') as table_count,
        (SELECT COUNT(*) FROM pg_indexes WHERE schemaname = 'public') as index_count;
    " >> "$LOG_FILE"
  done
fi

log "Maintenance tasks completed successfully!"

// ~/packages/node/pg-helper/index.js
const { Pool } = require('pg');

class PostgresHelper {
  constructor(config) {
    this.pool = new Pool(config);
    
    // Test connection
    this.pool.query('SELECT 1')
      .catch(err => console.error('Error connecting to PostgreSQL:', err));
  }
  
  async query(text, params) {
    const start = Date.now();
    const res = await this.pool.query(text, params);
    const duration = Date.now() - start;
    
    console.log('Executed query', { text, duration, rows: res.rowCount });
    return res;
  }
  
  async getClient() {
    const client = await this.pool.connect();
    const query = client.query;
    const release = client.release;
    
    // Set a timeout of 5 seconds on idle clients
    const timeout = setTimeout(() => {
      console.error('A client has been checked out for too long!');
      console.error(`The last executed query on this client was: ${client.lastQuery}`);
    }, 5000);
    
    // Monkey patch the query method to keep track of the last query executed
    client.query = (...args) => {
      client.lastQuery = args;
      return query.apply(client, args);
    };
    
    client.release = () => {
      clearTimeout(timeout);
      client.query = query;
      client.release = release;
      return release.apply(client);
    };
    
    return client;
  }
  
  async transaction(callback) {
    const client = await this.getClient();
    
    try {
      await client.query('BEGIN');
      const result = await callback(client);
      await client.query('COMMIT');
      return result;
    } catch (e) {
      await client.query('ROLLBACK');
      throw e;
    } finally {
      client.release();
    }
  }
  
  async close() {
    await this.pool.end();
  }
}

module.exports = PostgresHelper;

# ~/packages/python/pg_helper/pg_helper.py
import time
import logging
from contextlib import contextmanager
import psycopg2
from psycopg2.pool import SimpleConnectionPool
from psycopg2.extras import RealDictCursor

class PostgresHelper:
    def __init__(self, config):
        self.config = config
        self.pool = SimpleConnectionPool(1, 20, **config)
        self.logger = logging.getLogger('pg_helper')
        
        # Test connection
        with self.get_connection() as conn:
            with conn.cursor() as cursor:
                cursor.execute("SELECT 1")
    
    def get_connection(self):
        return self.pool.getconn()
    
    def return_connection(self, conn):
        self.pool.putconn(conn)
    
    @contextmanager
    def connection(self):
        conn = self.get_connection()
        try:
            yield conn
        finally:
            self.return_connection(conn)
    
    @contextmanager
    def cursor(self, cursor_factory=RealDictCursor):
        with self.connection() as conn:
            with conn.cursor(cursor_factory=cursor_factory) as cursor:
                yield cursor
    
    def query(self, query, params=None):
        start_time = time.time()
        with self.cursor() as cursor:
            cursor.execute(query, params)
            if cursor.description:
                result = cursor.fetchall()
            else:
                result = None
        
        duration = (time.time() - start_time) * 1000
        self.logger.debug(f"Query executed in {duration:.2f}ms: {query}")
        return result
    
    @contextmanager
    def transaction(self):
        with self.connection() as conn:
            try:
                conn.autocommit = False
                yield conn
                conn.commit()
            except Exception as e:
                conn.rollback()
                self.logger.error(f"Transaction rolled back: {str(e)}")
                raise
            finally:
                conn.autocommit = True
    
    def close(self):
        self.pool.closeall()

usage() {
  echo "PostgreSQL Issue Tracker"
  echo ""
  echo "Usage: $0 <command> [options]"
  echo ""
  echo "Commands:"
  echo "  list                List all open issues"
  echo "  create <title>      Create a new issue"

  echo "  update <id>         Update an issue"

  echo "  report              Generate diagnostic report"
  echo "  export              Export issues to JSON"
  echo "  help                Show this help message"
  echo ""
}

list_issues() {
  echo "PostgreSQL Issues:"
  echo "================="
  
  if [ ! "$(ls -A "$ISSUES_DIR")" ]; then
    echo "No issues found."
    return
  fi
  
  for issue in "$ISSUES_DIR"/*.issue; do
    id=$(basename "$issue" .issue)
    title=$(grep "^TITLE:" "$issue" | cut -d':' -f2- | xargs)
    status=$(grep "^STATUS:" "$issue" | cut -d':' -f2- | xargs)
    created=$(grep "^CREATED:" "$issue" | cut -d':' -f2- | xargs)
    
    if [ "$status" == "OPEN" ]; then
      echo "[$id] $title (Created: $created)"
    else
      echo "[$id] $title (Status: $status, Created: $created)"
    fi
  done
}

create_issue() {
  if [ $# -lt 1 ]; then
    echo "Error: Issue title required"
    usage
    exit 1
  fi
  
  title="$1"
  id=$(date +%Y%m%d%H%M%S)
  
  # Gather system information
  pg_version=$(sudo -u postgres psql -tAc "SELECT version();")
  os_info=$(lsb_release -ds)
  mem_info=$(free -h | grep Mem | awk '{print "Total: " $2 ", Used: " $3 ", Free: " $4}')
  disk_info=$(df -h / | tail -n 1 | awk '{print "Total: " $2 ", Used: " $3 ", Used%: " $5 ", Free: " $4}')
  
  # Create issue file
  cat > "$ISSUES_DIR/$id.issue" << EOF
TITLE:$title
STATUS:OPEN
CREATED:$(date '+%Y-%m-%d %H:%M:%S')
UPDATED:$(date '+%Y-%m-%d %H:%M:%S')

SYSTEM INFO:
PostgreSQL: $pg_version
OS: $os_info
Memory: $mem_info
Disk: $disk_info

DESCRIPTION:
[Add a detailed description here]

STEPS TO REPRODUCE:
1. 
2.
3.

SOLUTION:
[Not resolved yet]

NOTES:
EOF
  
  echo "Issue created with ID: $id"
  echo "Edit it with: $0 update $id"
}

show_issue() {
  if [ $# -lt 1 ]; then
    echo "Error: Issue ID required"
    usage
    exit 1
  fi
  
  id="$1"
  issue_file="$ISSUES_DIR/$id.issue"
  
  if [ ! -f "$issue_file" ]; then
    echo "Error: Issue $id not found"
    exit 1
  fi
  
  cat "$issue_file"
}

update_issue() {
  if [ $# -lt 1 ]; then
    echo "Error: Issue ID required"
    usage
    exit 1
  fi
  
  id="$1"
  issue_file="$ISSUES_DIR/$id.issue"

  if [ $# -lt 1 ]; then
    echo "Error: Issue ID required"
    usage
    exit 1
  fi

  issue_file="$ISSUES_DIR/$id.issue"

generate_report() {
  report_file="$ISSUES_DIR/report_$(date +%Y%m%d).txt"

  echo "PostgreSQL Configuration:" >> "$report_file"

  echo "" >> "$report_file"
  
  echo "Database Sizes:" >> "$report_file"
  sudo -u postgres psql -c "SELECT pg_database.datname, pg_size_pretty(pg_database_size(pg_database.datname)) FROM pg_database ORDER BY pg_database_size(pg_database.datname) DESC;" >> "$report_file"

  for issue in "$ISSUES_DIR"/*.issue; do
    if grep -q "^STATUS:OPEN" "$issue"; then
      cat "$issue" >> "$report_file"
      echo "----------------------------------------" >> "$report_file"
    fi
  done

      status=$(grep "^STATUS:" "$issue" | cut -d':' -f2- | xargs)
      created=$(grep "^CREATED:" "$issue" | cut -d':' -f2- | xargs)

    "updated": "$updated",
    "description": "$description",

  }
EOF
    fi
  done

  echo "Issues exported to: $export_file"
}

  log "Checking PostgreSQL status..."
  
  if systemctl is-active postgresql &>/dev/null; then
    echo -e "${GREEN}PostgreSQL service is running${NC}"
  else
    echo -e "${RED}PostgreSQL service is NOT running${NC}"
    return 1
  fi
  
  # Check if we can connect
  if sudo -u postgres psql -c "SELECT 1;" &>/dev/null; then
    echo -e "${GREEN}PostgreSQL connection successful${NC}"
  else
    echo -e "${RED}PostgreSQL connection failed${NC}"
    return 1
  fi
  
  return 0
}

# Check database sizes
check_sizes() {
  log "Checking database sizes..."
  
  echo -e "${BLUE}Database Sizes:${NC}"
  sudo -u postgres psql -c "SELECT pg_database.datname as database, 
    pg_size_pretty(pg_database_size(pg_database.datname)) as size,
    pg_database_size(pg_database.datname) as raw_size
    FROM pg_database 
    ORDER BY raw_size DESC;"
  
  # Check disk space
  echo -e "\n${BLUE}Disk Space:${NC}"
  df -h /var/lib/postgresql
}

# Check connections
check_connections() {
  log "Checking database connections..."
  
  # Get connection limits
  max_conn=$(sudo -u postgres psql -tAc "SHOW max_connections;")
  current_conn=$(sudo -u postgres psql -tAc "SELECT count(*) FROM pg_stat_activity;")
  pct_used=$((current_conn * 100 / max_conn))
  
  echo -e "${BLUE}Connection Usage:${NC}"
  echo "Current connections: $current_conn / $max_conn ($pct_used%)"
  
  if [ $pct_used -gt 80 ]; then
    echo -e "${RED}Warning: High connection usage!${NC}"
  elif [ $pct_used -gt 50 ]; then
    echo -e "${YELLOW}Notice: Moderate connection usage${NC}"
  else
    echo -e "${GREEN}Connection usage is normal${NC}"
  fi
  
  # Show connection details
  echo -e "\n${BLUE}Connection Details:${NC}"
  sudo -u postgres psql -c "SELECT datname as database, 
    usename as user, 
    application_name, 
    client_addr, 
    state, 
    count(*) 
    FROM pg_stat_activity 
    GROUP BY 1, 2, 3, 4, 5 
    ORDER BY count DESC;"
}

# Check for long-running queries
check_long_queries() {
  log "Checking for long-running queries..."
  
  echo -e "${BLUE}Long-Running Queries:${NC}"
  sudo -u postgres psql -c "SELECT pid, 
    now() - pg_stat_activity.query_start AS duration, 
    usename, 
    datname, 
    wait_event, 
    wait_event_type,
    substr(query, 1, 80) as query_snippet
    FROM pg_stat_activity 
    WHERE state = 'active' 
    AND (now() - pg_stat_activity.query_start) > interval '10 seconds'
    ORDER BY duration DESC;"
}

# Check for locks
check_locks() {
  log "Checking for locks..."
  
  echo -e "${BLUE}Database Locks:${NC}"
  sudo -u postgres psql -c "SELECT locktype, 
    relation::regclass, 
    mode, 
    transactionid as tid, 
    virtualtransaction as vtid, 
    pid, 
    granted 
    FROM pg_locks l 
    LEFT JOIN pg_stat_activity a ON l.pid = a.pid 
    ORDER BY relation;"
  
  # Check for blocking locks
  echo -e "\n${BLUE}Blocking Locks:${NC}"
  sudo -u postgres psql -c "SELECT blocked_activity.pid AS blocked_pid,
    blocked_activity.usename AS blocked_user,
    blocking_activity.pid AS blocking_pid,
    blocking_activity.usename AS blocking_user,
    blocked_activity.query AS blocked_statement,
    blocking_activity.query AS blocking_statement
    FROM pg_catalog.pg_locks blocked_locks
    JOIN pg_catalog.pg_stat_activity blocked_activity ON blocked_activity.pid = blocked_locks.pid
    JOIN pg_catalog.pg_locks blocking_locks ON blocking_locks.locktype = blocked_locks.locktype
      AND blocking_locks.database IS NOT DISTINCT FROM blocked_locks.database
      AND blocking_locks.relation IS NOT DISTINCT FROM blocked_locks.relation
      AND blocking_locks.page IS NOT DISTINCT FROM blocked_locks.page
      AND blocking_locks.tuple IS NOT DISTINCT FROM blocked_locks.tuple
      AND blocking_locks.virtualxid IS NOT DISTINCT FROM blocked_locks.virtualxid
      AND blocking_locks.transactionid IS NOT DISTINCT FROM blocked_locks.transactionid
      AND blocking_locks.classid IS NOT DISTINCT FROM blocked_locks.classid
      AND blocking_locks.objid IS NOT DISTINCT FROM blocked_locks.objid
      AND blocking_locks.objsubid IS NOT DISTINCT FROM blocked_locks.objsubid
      AND blocking_locks.pid != blocked_locks.pid
    JOIN pg_catalog.pg_stat_activity blocking_activity ON blocking_activity.pid = blocking_locks.pid
    WHERE NOT blocked_locks.granted AND blocking_locks.granted;"
}

# Check for index usage
check_indexes() {
  log "Checking index usage..."
  
  echo -e "${BLUE}Index Usage Statistics:${NC}"
  sudo -u postgres psql -c "SELECT schemaname, 
    relname, 
    indexrelname, 
    idx_scan, 
    idx_tup_read, 
    idx_tup_fetch 
    FROM pg_stat_user_indexes 
    JOIN pg_stat_user_tables ON pg_stat_user_indexes.relid = pg_stat_user_tables.relid 
    ORDER BY idx_scan DESC 
    LIMIT 20;"
  
  # Check for unused indexes
  echo -e "\n${BLUE}Unused Indexes:${NC}"
  sudo -u postgres psql -c "SELECT schemaname, 
    relname, 
    indexrelname, 
    idx_scan, 
    pg_size_pretty(pg_relation_size(indexrelname::regclass)) as index_size
    FROM pg_stat_user_indexes 
    JOIN pg_stat_user_tables ON pg_stat_user_indexes.relid = pg_stat_user_tables.relid 
    WHERE idx_scan = 0 
    AND pg_stat_user_tables.n_live_tup > 100
    AND pg_relation_size(indexrelname::regclass) > 1024 * 1024
    ORDER BY pg_relation_size(indexrelname::regclass) DESC;"
}

# Check table statistics
check_tables() {
  log "Checking table statistics..."
  
  echo -e "${BLUE}Table Access Statistics:${NC}"
  sudo -u postgres psql -c "SELECT schemaname, 
    relname, 
    seq_scan, 
    seq_tup_read, 
    idx_scan, 
    idx_tup_fetch, 
    n_tup_ins, 
    n_tup_upd, 
    n_tup_del, 
    n_live_tup, 
    n_dead_tup, 
    last_vacuum, 
    last_autovacuum, 
    last_analyze, 
    last_autoanalyze 
    FROM pg_stat_user_tables 
    ORDER BY n_live_tup DESC 
    LIMIT 10;"
  
  # Check for tables needing vacuum
  echo -e "\n${BLUE}Tables Needing VACUUM:${NC}"
  sudo -u postgres psql -c "SELECT schemaname, 
    relname, 
    n_live_tup, 
    n_dead_tup, 
    n_dead_tup * 100.0 / GREATEST(n_live_tup, 1) as dead_ratio,
    last_vacuum, 
    last_autovacuum 
    FROM pg_stat_user_tables 
    WHERE n_dead_tup > 10000 OR n_dead_tup > n_live_tup * 0.1
    ORDER BY n_dead_tup DESC 
    LIMIT 10;"
}

# Check replication status
check_replication() {
  log "Checking replication status..."
  
  # Check if this instance is a primary
  is_primary=$(sudo -u postgres psql -tAc "SELECT pg_is_in_recovery();")
  
  if [ "$is_primary" = "f" ]; then
    echo -e "${BLUE}This is a PRIMARY server${NC}"
    
    echo -e "\n${BLUE}Replication Status:${NC}"
    sudo -u postgres psql -c "SELECT client_addr, 
      usename, 
      application_name, 
      state, 
      sent_lsn, 
      write_lsn, 
      flush_lsn, 
      replay_lsn, 
      (sent_lsn - replay_lsn) AS lag_bytes,
      sync_state 
      FROM pg_stat_replication;"
    
    echo -e "\n${BLUE}Replication Slots:${NC}"
    sudo -u postgres psql -c "SELECT slot_name, 
      plugin, 
      slot_type, 
      datoid, 
      active, 
      xmin, 
      pg_current_wal_lsn() - restart_lsn AS retained_bytes,
      pg_size_pretty(pg_current_wal_lsn() - restart_lsn) AS retained_size
      FROM pg_replication_slots;"
  else
    echo -e "${BLUE}This is a REPLICA server${NC}"
    
    echo -e "\n${BLUE}Replication Status:${NC}"
    sudo -u postgres psql -c "SELECT pg_last_wal_receive_lsn(), 
      pg_last_wal_replay_lsn(), 
      pg_last_wal_receive_lsn() - pg_last_wal_replay_lsn() AS lag_bytes,
      pg_size_pretty(pg_last_wal_receive_lsn() - pg_last_wal_replay_lsn()) AS lag_size,
      pg_last_xact_replay_timestamp(), 
      CASE WHEN pg_last_wal_receive_lsn() = pg_last_wal_replay_lsn() THEN 0
      ELSE EXTRACT(EPOCH FROM now() - pg_last_xact_replay_timestamp())
      END AS lag_seconds;"
  fi
}

# Check cache hit ratio
check_cache() {
  log "Checking cache hit ratios..."
  
  echo -e "${BLUE}Buffer Cache Hit Ratio:${NC}"
  sudo -u postgres psql -c "SELECT 
    sum(heap_blks_read) as heap_read,
    sum(heap_blks_hit) as heap_hit,
    case when sum(heap_blks_read) = 0 then 'N/A'
    else (sum(heap_blks_hit) * 100 / (sum(heap_blks_hit) + sum(heap_blks_read)))::text
    end as hit_ratio
    FROM pg_statio_user_tables;"
  
  echo -e "\n${BLUE}Index Cache Hit Ratio:${NC}"
  sudo -u postgres psql -c "SELECT 
    sum(idx_blks_read) as idx_read,
    sum(idx_blks_hit) as idx_hit,
    case when sum(idx_blks_read) = 0 then 'N/A'
    else (sum(idx_blks_hit) * 100 / (sum(idx_blks_hit) + sum(idx_blks_read)))::text
    end as hit_ratio
    FROM pg_statio_user_indexes;"
}

# Check slow queries
check_slow_queries() {
  log "Checking slow query statistics..."
  
  # Check if pg_stat_statements is installed
  stmt_exists=$(sudo -u postgres psql -tAc "SELECT EXISTS(SELECT 1 FROM pg_extension WHERE extname = 'pg_stat_statements');")
  
  if [ "$stmt_exists" = "t" ]; then
    echo -e "${BLUE}Top 10 Slowest Queries:${NC}"
    sudo -u postgres psql -c "SELECT 
      substring(query, 1, 80) as query_snippet,
      calls, 
      round(total_exec_time::numeric, 2) as total_time_ms,
      round(mean_exec_time::numeric, 2) as avg_time_ms,
      round((100 * total_exec_time / sum(total_exec_time) OVER ())::numeric, 2) as percent,
      round(stddev_exec_time::numeric, 2) as stddev_ms,
      round(min_exec_time::numeric, 2) as min_ms,
      round(max_exec_time::numeric, 2) as max_ms
      FROM pg_stat_statements 
      ORDER BY total_exec_time DESC 
      LIMIT 10;"
    
    echo -e "\n${BLUE}Top 10 IO-Intensive Queries:${NC}"
    sudo -u postgres psql -c "SELECT 
      substring(query, 1, 80) as query_snippet,
      calls, 
      shared_blks_hit,
      shared_blks_read,
      shared_blks_dirtied,
      shared_blks_written,
      local_blks_hit,
      local_blks_read,
      temp_blks_read,
      temp_blks_written
      FROM pg_stat_statements 
      ORDER BY shared_blks_read + shared_blks_written + temp_blks_read + temp_blks_written DESC 
      LIMIT 10;"
  else
    echo -e "${YELLOW}pg_stat_statements extension is not installed. Enable it to see query statistics.${NC}"
    echo "Run: sudo -u postgres psql -c \"CREATE EXTENSION pg_stat_statements;\""
    echo "And add to postgresql.conf: shared_preload_libraries = 'pg_stat_statements'"
  fi
}

# Check for bloat
check_bloat() {
  log "Checking for table and index bloat..."
  
  echo -e "${BLUE}Table Bloat Estimation:${NC}"
  sudo -u postgres psql -c "SELECT 
    schemaname, 
    tablename, 
    cc.reltuples as row_estimate,
    pg_size_pretty(pg_total_relation_size('"'\"' || schemaname || '"'.\"' || tablename || '"'\"')) as total_size,
    pg_size_pretty(pg_relation_size('"'\"' || schemaname || '"'.\"' || tablename || '"'\"')) as table_size,
    pg_size_pretty(pg_total_relation_size('"'\"' || schemaname || '"'.\"' || tablename || '"'\"') - pg_relation_size('"'\"' || schemaname || '"'.\"' || tablename || '"'\"')) as index_size,
    CASE WHEN pg_total_relation_size('"'\"' || schemaname || '"'.\"' || tablename || '"'\"') > 0
    THEN round(100 * (pg_total_relation_size('"'\"' || schemaname || '"'.\"' || tablename || '"'\"') - pg_relation_size('"'\"' || schemaname || '"'.\"' || tablename || '"'\"')) / pg_total_relation_size('"'\"' || schemaname || '"'.\"' || tablename || '"'\"'))
    ELSE 0
    END as index_pct
    FROM pg_tables t
    JOIN pg_class cc ON t.tablename = cc.relname
    WHERE t.schemaname NOT IN ('pg_catalog', 'information_schema')
    AND pg_total_relation_size('"'\"' || schemaname || '"'.\"' || tablename || '"'\"') > 10 * 1024 * 1024
    ORDER BY pg_total_relation_size('"'\"' || schemaname || '"'.\"' || tablename || '"'\"') DESC
    LIMIT 10;"
  
  echo -e "\n${BLUE}Most Bloated Tables:${NC}"
  sudo -u postgres psql -c "WITH RECURSIVE constants (vacuum_threshold) as (
    values (20)
    ),
    table_bloat as (
    select
      schemaname, tablename, reltuples,
      pg_table_size(schemaname || '.' || tablename) as table_size,
      pg_indexes_size(schemaname || '.' || tablename) as index_size,
      pg_total_relation_size(schemaname || '.' || tablename) as total_size,
      n_live_tup, n_dead_tup,
      case when n_live_tup = 0 then 0 else round((n_dead_tup::float / n_live_tup::float) * 100) end as dead_pct
    from pg_stat_user_tables
    join pg_tables using (schemaname, tablename)
    join pg_class on relname = tablename
    ),
    bloat_summary as (
      select 
        schemaname, tablename, reltuples, 
        table_size, index_size, total_size,
        pg_size_pretty(table_size) as pretty_table_size,
        pg_size_pretty(index_size) as pretty_index_size,
        pg_size_pretty(total_size) as pretty_total_size,
        n_live_tup, n_dead_tup, dead_pct
      from table_bloat
    )
    select * from bloat_summary 
    where dead_pct > (select vacuum_threshold from constants)
    and n_live_tup > 1000
    order by dead_pct desc limit 10;"
}

# Check transaction wraparound
check_txid() {
  log "Checking transaction ID wraparound..."
  
  echo -e "${BLUE}Transaction ID Wraparound Status:${NC}"
  sudo -u postgres psql -c "SELECT 
    datname, 
    age(datfrozenxid) as xid_age,
    round(100*(2147483647-age(datfrozenxid))/2147483647.0, 4) as pct_remaining,
    pg_size_pretty(pg_database_size(datname)) as db_size
    FROM pg_database 
    ORDER BY age(datfrozenxid) DESC;"
  
  # Warn if any database is approaching wraparound
  max_age=$(sudo -u postgres psql -tAc "SELECT max(age(datfrozenxid)) FROM pg_database;")
  warn_threshold=1000000000  # About halfway to wraparound
  
  if [ "$max_age" -gt $warn_threshold ]; then
    echo -e "${RED}WARNING: Database(s) are approaching transaction ID wraparound!${NC}"
    echo "Consider running VACUUM FREEZE on affected databases."
  else
    echo -e "${GREEN}Transaction ID usage is normal${NC}"
  fi
}
