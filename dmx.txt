5. GitHub Integration
Add a GitHub integration script to ~/dotfiles/tools/:

bash
#!/usr/bin/env bash
# ~/dotfiles/tools/gh_setup.sh
set -euo pipefail

# Default repository owner
OWNER="peteywee"

# Parse arguments
while [[ $# -gt 0 ]]; do
  case "$1" in
    --repo)
      REPO="$2"
      shift 2
      ;;
    --public)
      PUBLIC=true
      shift
      ;;
    --description)
      DESCRIPTION="$2"
      shift 2
      ;;
    *)
      echo "Unknown option: $1"
      exit 1
      ;;
  esac
done

# Check if GitHub CLI is installed
if ! command -v gh &>/dev/null; then
  echo "Installing GitHub CLI..."
  sudo apt-key adv --keyserver keyserver.ubuntu.com --recv-key C99B11DEB97541F0
  sudo apt-add-repository https://cli.github.com/packages
  sudo apt update
  sudo apt install gh
fi

# Check if authenticated
if ! gh auth status &>/dev/null; then
  echo "Please authenticate with GitHub:"
  gh auth login
fi

# Create repository
if [[ -n "${REPO:-}" ]]; then
  VISIBILITY=${PUBLIC:+--public}
  VISIBILITY=${VISIBILITY:---private}
  DESC=${DESCRIPTION:+--description "${DESCRIPTION}"}
  
  echo "Creating repository ${OWNER}/${REPO}..."
  gh repo create "${OWNER}/${REPO}" ${VISIBILITY} ${DESC} --confirm
  
  # Clone the repository if it doesn't exist locally
  if [[ ! -d "${REPO}" ]]; then
    git clone "https://github.com/${OWNER}/${REPO}.git"
  fi
  
  echo "Repository created: https://github.com/${OWNER}/${REPO}"
else
  echo "Usage: gh_setup.sh --repo REPO_NAME [--public] [--description \"Repo description\"]"
  exit 1
fi

6. Knowledge Base System
Add this script to manage your knowledge base:

bash
#!/usr/bin/env bash
# ~/knowledge_base/scripts/kb_manager.sh
set -euo pipefail

KB_DIR="$HOME/knowledge_base"
KB_DB="$KB_DIR/database/kb.sqlite"

# Ensure directory structure exists
mkdir -p "$KB_DIR"/{raw,docs,metadata,database,scripts,tools}

# Create SQLite database if it doesn't exist
if [[ ! -f "$KB_DB" ]]; then
  echo "Creating knowledge base database..."
  sqlite3 "$KB_DB" <<EOF
CREATE TABLE projects (
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL,
  description TEXT,
  is_private INTEGER DEFAULT 1,
  created_at TEXT DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE documents (
  id INTEGER PRIMARY KEY,
  project_id INTEGER,
  filename TEXT NOT NULL,
  content TEXT,
  created_at TEXT DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (project_id) REFERENCES projects (id)
);

CREATE TABLE tags (
  id INTEGER PRIMARY KEY,
  name TEXT UNIQUE NOT NULL
);

CREATE TABLE document_tags (
  document_id INTEGER,
  tag_id INTEGER,
  PRIMARY KEY (document_id, tag_id),
  FOREIGN KEY (document_id) REFERENCES documents (id),
  FOREIGN KEY (tag_id) REFERENCES tags (id)
);
EOF
fi

# Function to add a project
add_project() {
  local name="$1"
  local description="${2:-}"
  local is_private="${3:-1}"
  
  sqlite3 "$KB_DB" <<EOF
INSERT INTO projects (name, description, is_private)
VALUES ('$name', '$description', $is_private);
EOF
  
  echo "Project '$name' added to knowledge base."
}

# Function to add a document
add_document() {
  local project_name="$1"
  local filename="$2"
  local content_file="$3"
  
  # Get project ID
  local project_id=$(sqlite3 "$KB_DB" "SELECT id FROM projects WHERE name = '$project_name';")
  
  if [[ -z "$project_id" ]]; then
    echo "Project '$project_name' not found. Creating it..."
    add_project "$project_name"
    project_id=$(sqlite3 "$KB_DB" "SELECT id FROM projects WHERE name = '$project_name';")
  fi
  
  # Read content file
  if [[ -f "$content_file" ]]; then
    # Escape single quotes for SQL
    local content=$(cat "$content_file" | sed "s/'/''/g")
    
    sqlite3 "$KB_DB" <<EOF
INSERT INTO documents (project_id, filename, content)
VALUES ($project_id, '$filename', '$content');
EOF
    
    echo "Document '$filename' added to project '$project_name'."
  else
    echo "Content file '$content_file' not found."
    exit 1
  fi
}

# Function to add a tag to a document
add_tag() {
  local project_name="$1"
  local filename="$2"
  local tag_name="$3"
  
  # Get document ID
  local document_id=$(sqlite3 "$KB_DB" <<EOF
SELECT d.id FROM documents d
JOIN projects p ON d.project_id = p.id
WHERE p.name = '$project_name' AND d.filename = '$filename';
EOF
)
  
  if [[ -z "$document_id" ]]; then
    echo "Document '$filename' in project '$project_name' not found."
    exit 1
  fi
  
  # Get tag ID or create it
  local tag_id=$(sqlite3 "$KB_DB" "SELECT id FROM tags WHERE name = '$tag_name';")
  
  if [[ -z "$tag_id" ]]; then
    sqlite3 "$KB_DB" "INSERT INTO tags (name) VALUES ('$tag_name');"
    tag_id=$(sqlite3 "$KB_DB" "SELECT id FROM tags WHERE name = '$tag_name';")
  fi
  
  # Add tag to document
  sqlite3 "$KB_DB" <<EOF
INSERT OR IGNORE INTO document_tags (document_id, tag_id)
VALUES ($document_id, $tag_id);
EOF
  
  echo "Tag '$tag_name' added to document '$filename'."
}

# Function to search knowledge base
search_kb() {
  local query="$1"
  local tag="${2:-}"
  
  local tag_clause=""
  if [[ -n "$tag" ]]; then
    tag_clause="AND t.name = '$tag'"
  fi
  
  sqlite3 "$KB_DB" <<EOF
.mode column
.headers on
SELECT 
  p.name AS Project,
  d.filename AS Filename,
  substr(d.content, 1, 50) || '...' AS Preview
FROM documents d
JOIN projects p ON d.project_id = p.id
LEFT JOIN document_tags dt ON d.id = dt.document_id
LEFT JOIN tags t ON dt.tag_id = t.id
WHERE d.content LIKE '%$query%'
$tag_clause
ORDER BY p.name, d.filename;
EOF
}

    echo "  kb_manager.sh add-document PROJECT FILENAME CONTENT_FILE"
    echo "  kb_manager.sh add-tag PROJECT FILENAME TAG"

8. Maintenance & Updates
Create a watchdog service that monitors and maintains your environment:

bash
#!/usr/bin/env bash
# ~/.assistant/watchdog/monitor.sh
set -euo pipefail

LOG_DIR="$HOME/.assistant/logs"
mkdir -p "$LOG_DIR"
LOG_FILE="$LOG_DIR/monitor.log"

log() {
  echo "[$(date '+%Y-%m-%d %H:%M:%S')] $*" | tee -a "$LOG_FILE"
}

check_system() {
  log "Checking system health..."
  
  # Check disk space
  df_output=$(df -h / | tail -n 1)
  used_percent=$(echo "$df_output" | awk '{print $5}' | tr -d '%')
  
  if (( used_percent > 90 )); then
    log "WARNING: Disk space critical ($used_percent% used)"
  fi
  
  # Check memory
  free_output=$(free -m | grep Mem)
  total_mem=$(echo "$free_output" | awk '{print $2}')
  used_mem=$(echo "$free_output" | awk '{print $3}')
  used_percent=$((used_mem * 100 / total_mem))
  
  if (( used_percent > 90 )); then
    log "WARNING: Memory usage high ($used_percent%)"
  fi
  
  # Check color system
  if [[ ! -d "$HOME/.color_system" ]]; then
    log "ERROR: Color system not found, attempting repair"
    "$HOME/dotfiles/bootstrap.sh"
  fi
  
  # Check knowledge base
  if [[ ! -f "$HOME/knowledge_base/database/kb.sqlite" ]]; then
    log "ERROR: Knowledge base database not found, attempting repair"
    "$HOME/knowledge_base/scripts/kb_manager.sh" add-project "default" "Default project"
  fi
}

backup_config() {
  log "Creating configuration backup..."
  
  backup_dir="$HOME/backups/$(date '+%Y%m%d')"
  mkdir -p "$backup_dir"
  
  # Backup dotfiles
  if [[ -d "$HOME/dotfiles" ]]; then
    tar -czf "$backup_dir/dotfiles.tar.gz" -C "$HOME" dotfiles
  fi
  
  # Backup color system
  if [[ -d "$HOME/.color_system" ]]; then
    tar -czf "$backup_dir/color_system.tar.gz" -C "$HOME" .color_system
  fi
  
  # Backup knowledge base
  if [[ -d "$HOME/knowledge_base" ]]; then
    sqlite3 "$HOME/knowledge_base/database/kb.sqlite" .dump > "$backup_dir/kb_dump.sql"
  fi
  
  log "Backup created at $backup_dir"
}

update_system() {
  log "Checking for updates..."
  
  # Update dotfiles
  if [[ -d "$HOME/dotfiles/.git" ]]; then
    cd "$HOME/dotfiles"
    git fetch origin
    LOCAL=$(git rev-parse @)
    REMOTE=$(git rev-parse @{u})
    
    if [[ "$LOCAL" != "$REMOTE" ]]; then
      log "Updates available for dotfiles, pulling changes"
      git pull
      "$HOME/dotfiles/bootstrap.sh"
    fi
  fi
  
  # Check for system updates
  sudo apt update
  updates=$(apt list --upgradable 2>/dev/null | grep -c upgradable)
  
  if (( updates > 0 )); then
    log "$updates package updates available"
  fi
}

case "${1:-check}" in
  check)
    check_system
    ;;
  backup)
    backup_config
    ;;
  update)
    update_system
    ;;
  full)
    check_system
    backup_config
    update_system
    ;;
  *)
    echo "Usage: monitor.sh [check|backup|update|full]"
    exit 1
    ;;
esac

log "Monitor completed successfully"

#!/usr/bin/env bash
set -euo pipefail

LOG_DIR="$HOME/.logs/watchdog"
mkdir -p "$LOG_DIR"
LOG_FILE="$LOG_DIR/$(date +%Y%m%d).log"

log() {
  echo "[$(date '+%Y-%m-%d %H:%M:%S')] $*" | tee -a "$LOG_FILE"
}

check_service() {
  local service="$1"
  if ! systemctl --user is-active "$service" &>/dev/null; then
    log "Service $service is not running. Attempting to restart..."
    systemctl --user restart "$service" || log "Failed to restart $service"
  else
    log "Service $service is running."
  fi
}

check_process() {
  local process="$1"
  if ! pgrep -f "$process" &>/dev/null; then
    log "Process $process is not running."
  else
    log "Process $process is running."
  fi
}

check_disk_space() {
  local threshold=90
  local usage=$(df -h / | tail -n 1 | awk '{print $5}' | tr -d '%')
  
  if (( usage > threshold )); then
    log "WARNING: Disk space critical ($usage% used)"
  else
    log "Disk space usage: $usage%"
  fi
}

check_memory() {
  local threshold=90
  local free_output=$(free -m | grep Mem)
  local total_mem=$(echo "$free_output" | awk '{print $2}')
  local used_mem=$(echo "$free_output" | awk '{print $3}')
  local used_percent=$((used_mem * 100 / total_mem))
  
  if (( used_percent > threshold )); then
    log "WARNING: Memory usage high ($used_percent%)"
  else
    log "Memory usage: $used_percent%"
  fi
}

check_postgres() {
  if systemctl is-active postgresql &>/dev/null; then
    log "PostgreSQL is running."
  else
    log "WARNING: PostgreSQL is not running. Attempting to start..."
    sudo systemctl start postgresql || log "Failed to start PostgreSQL"
  fi
}

check_docker() {
  if systemctl is-active docker &>/dev/null; then
    log "Docker is running."
  else
    log "WARNING: Docker is not running. Attempting to start..."
    sudo systemctl start docker || log "Failed to start Docker"
  fi
}

  log "=== Starting system checks ==="
  
  # Check services
  check_service "postgres-backup.timer"
  
  # Check processes
  check_process "postgres"
  check_process "docker"
  
  # Check system resources
  check_disk_space
  check_memory
  
  # Check database
  check_postgres
  
  # Check Docker
  check_docker
  
  log "=== System checks completed ==="
}

cat > "$SERVICES_DIR/backup/backup.sh" << 'EOF'
#!/usr/bin/env bash
set -euo pipefail

BACKUP_ROOT="$HOME/backups"
LOG_DIR="$HOME/.logs/backup"
mkdir -p "$BACKUP_ROOT" "$LOG_DIR"
LOG_FILE="$LOG_DIR/$(date +%Y%m%d).log"

log() {
  echo "[$(date '+%Y-%m-%d %H:%M:%S')] $*" | tee -a "$LOG_FILE"
}

backup_dotfiles() {
  log "Backing up dotfiles..."
  mkdir -p "$BACKUP_ROOT/dotfiles"
  BACKUP_FILE="$BACKUP_ROOT/dotfiles/dotfiles_$(date +%Y%m%d_%H%M%S).tar.gz"
  
  tar -czf "$BACKUP_FILE" -C "$HOME" dotfiles
  log "Dotfiles backed up to $BACKUP_FILE"
}

backup_projects() {
  log "Backing up projects..."
  mkdir -p "$BACKUP_ROOT/projects"
  BACKUP_FILE="$BACKUP_ROOT/projects/projects_$(date +%Y%m%d_%H%M%S).tar.gz"
  
  find "$HOME/projects" -maxdepth 1 -type d -not -path "$HOME/projects" | while read -r project; do
    if [ -d "$project/.git" ]; then
      log "Backing up project: $(basename "$project")"
      tar -czf "$BACKUP_ROOT/projects/$(basename "$project")_$(date +%Y%m%d_%H%M%S).tar.gz" -C "$HOME/projects" "$(basename "$project")"
    fi
  done
  
  log "Projects backed up to $BACKUP_ROOT/projects"
}

backup_postgres() {
  log "Backing up PostgreSQL databases..."
  mkdir -p "$BACKUP_ROOT/postgres"
  
  if command -v pg_dumpall &>/dev/null; then
    pg_dumpall -c > "$BACKUP_ROOT/postgres/all_dbs_$(date +%Y%m%d_%H%M%S).sql"
    log "All PostgreSQL databases backed up"
  else
    log "pg_dumpall not found, skipping PostgreSQL backup"
  fi
}

clean_old_backups() {
  log "Cleaning up old backups..."
  
  # Keep backups for 30 days
  find "$BACKUP_ROOT" -type f -mtime +30 -name "*.tar.gz" -delete
  find "$BACKUP_ROOT" -type f -mtime +30 -name "*.sql" -delete
  
  log "Old backups cleaned up"
}

run_backup() {
  log "=== Starting backup ==="
  
  backup_dotfiles
  backup_projects
  backup_postgres
  clean_old_backups
  
  log "=== Backup completed ==="
}

repair_postgres() {
  log "Repairing PostgreSQL..."
  
  if ! systemctl is-active postgresql &>/dev/null; then
    log "Starting PostgreSQL service..."
    sudo systemctl start postgresql
  fi
  
  log "Checking PostgreSQL connections..."
  if ! psql -c "SELECT 1;" &>/dev/null; then
    log "PostgreSQL connection failed, attempting repair..."
    sudo systemctl restart postgresql
  fi
}

repair_docker() {
  log "Repairing Docker..."
  
  if ! systemctl is-active docker &>/dev/null; then
    log "Starting Docker service..."
    sudo systemctl start docker
  fi
  
  log "Checking Docker daemon..."
  if ! docker info &>/dev/null; then
    log "Docker daemon not responding, attempting repair..."
    sudo systemctl restart docker
  fi
}

repair_dotfiles() {
  log "Checking dotfiles integrity..."
  
  if [ ! -d "$HOME/dotfiles" ]; then
    log "Dotfiles directory not found, attempting to restore from backup..."
    LATEST_BACKUP=$(find "$HOME/backups/dotfiles" -name "dotfiles_*.tar.gz" -type f | sort | tail -n 1)
    
    if [ -n "$LATEST_BACKUP" ]; then
      log "Restoring dotfiles from $LATEST_BACKUP..."
      tar -xzf "$LATEST_BACKUP" -C "$HOME"
    else
      log "No dotfiles backup found"
    fi
  fi
  
  if [ ! -f "$HOME/.bashrc" ]; then
    log "Restoring .bashrc..."
    ln -sf "$HOME/dotfiles/bash/.bashrc" "$HOME/.bashrc"
  fi
}

repair_services() {
  log "Checking systemd user services..."
  
  systemctl --user daemon-reload
  
  for service in watchdog.timer backup.timer scheduler.timer postgres-backup.timer; do
    if ! systemctl --user is-enabled "$service" &>/dev/null; then
      log "Enabling service: $service..."
      systemctl --user enable "$service"
    fi
    
    if ! systemctl --user is-active "$service" &>/dev/null; then
      log "Starting service: $service..."
      systemctl --user start "$service"
    fi
  done
}

run_self_healing() {
  log "=== Starting self-healing process ==="
  
  repair_postgres
  repair_docker
  repair_dotfiles
  repair_services
  
  log "=== Self-healing completed ==="
}

run_self_healing

        if [[ -z "\$text" ]]; then
          jq -r ".roles.\${role}.code" "\$COLOR_ROLES_FILE" 2>/dev/null || echo ""
          return
        fi
        
        # Get color code, apply to text, and reset
        local color_code=\$(jq -r ".roles.\${role}.code" "\$COLOR_ROLES_FILE" 2>/dev/null)
        local reset_code=\$(jq -r ".roles.reset.code" "\$COLOR_ROLES_FILE" 2>/dev/null)
        
        if [[ -n "\$color_code" && -n "\$reset_code" ]]; then
          echo -e "\${color_code}\${text}\${reset_code}"
        else
          echo "\$text"
        fi
      }
      EOF2
      
      # Create aliases
      cat > "${INSTALL_DIR}/aliases.sh" << EOF2
      #!/usr/bin/env bash
      # Color Role System Aliases
      
      alias cwarn='color_apply warning'
      alias cerror='color_apply error'
      alias csuccess='color_apply success' 
      alias cinfo='color_apply info'
      alias creset='color_apply reset'
      EOF2
      
      # Add to .bashrc
      if ! grep -q "Color Role System" "$BASHRC"; then
        echo "
      # Color Role System
      source \"${INSTALL_DIR}/color_roles_loader.sh\"
      source \"${INSTALL_DIR}/aliases.sh\"" >> "$BASHRC"
      fi
      
      chmod +x "${INSTALL_DIR}/color_roles_loader.sh"
      chmod +x "${INSTALL_DIR}/aliases.sh"
      
      echo "✅ Color Role System installed successfully."
    EOT
  }

# Check if Terraform is installed
check_terraform() {
  if ! command -v terraform &>/dev/null; then
    echo -e "${RED}Terraform is not installed.${NC}"
    echo "Installing Terraform..."
    
    # Add HashiCorp GPG key
    wget -O- https://apt.releases.hashicorp.com/gpg | \
      gpg --dearmor | \
      sudo tee /usr/share/keyrings/hashicorp-archive-keyring.gpg >/dev/null
    
    # Add HashiCorp repository
    echo "deb [signed-by=/usr/share/keyrings/hashicorp-archive-keyring.gpg] \
      https://apt.releases.hashicorp.com $(lsb_release -cs) main" | \
      sudo tee /etc/apt/sources.list.d/hashicorp.list
    
    # Install Terraform
    sudo apt-get update && sudo apt-get install -y terraform
  fi
}

# Check if Terragrunt is installed
check_terragrunt() {
  if ! command -v terragrunt &>/dev/null; then
    echo -e "${YELLOW}Terragrunt is not installed.${NC}"
    echo "Installing Terragrunt..."
    TERRAGRUNT_VERSION=$(curl -s https://api.github.com/repos/gruntwork-io/terragrunt/releases/latest | grep tag_name | cut -d '"' -f 4)
    
    # Download and install Terragrunt
    curl -Lo terragrunt "https://github.com/gruntwork-io/terragrunt/releases/download/${TERRAGRUNT_VERSION}/terragrunt_linux_amd64"
    chmod +x terragrunt
    sudo mv terragrunt /usr/local/bin/
  fi
}
